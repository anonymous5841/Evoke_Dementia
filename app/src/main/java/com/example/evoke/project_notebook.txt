package com.example.evoke.database

import android.content.Context
import androidx.room.Database
import androidx.room.Room
import androidx.room.RoomDatabase
import androidx.room.migration.Migration
import androidx.sqlite.db.SupportSQLiteDatabase
import com.example.evoke.security.DatabaseKeyManager
import net.zetetic.database.sqlcipher.SupportOpenHelperFactory
import java.io.File
import java.util.concurrent.Callable

@Database(
    entities = [
        User::class,
        Person::class,
        FaceEmbedding::class,
        VoiceEmbedding::class,
        UserVoiceEmbedding::class,
        RecognitionEvent::class,
        Conversation::class,
        PersonHistory::class,
        PendingConversation::class,
        Location::class,
        EmergencyContact::class
    ],
    version = 3,
    exportSchema = false
)
abstract class AppDatabase : RoomDatabase() {

    abstract fun userDao(): UserDao
    abstract fun personDao(): PersonDao
    abstract fun faceEmbeddingDao(): FaceEmbeddingDao
    abstract fun voiceEmbeddingDao(): VoiceEmbeddingDao
    abstract fun userVoiceEmbeddingDao(): UserVoiceEmbeddingDao
    abstract fun recognitionEventDao(): RecognitionEventDao
    abstract fun conversationDao(): ConversationDao
    abstract fun personHistoryDao(): PersonHistoryDao
    abstract fun pendingConversationDao(): PendingConversationDao
    abstract fun locationDao(): LocationDao
    abstract fun emergencyContactDao(): EmergencyContactDao

    companion object {

        private const val DATABASE_NAME = "evoke_database"
        private const val QUIESCE_ATTEMPTS = 3

        fun isOpen(): Boolean = INSTANCE != null

        @Volatile
        private var INSTANCE: AppDatabase? = null

        @Volatile
        private var nativeLibraryLoaded = false

        // Version 1 -> 2: audio path and processing status for pending conversations.
        //
        // NOTE: this only works if the v1 pending_conversations table has no
        // column that the current entity no longer has (for example an old raw
        // conversation text column). If it did, recreate the table instead.
        private val MIGRATION_1_2 = object : Migration(1, 2) {

            override fun migrate(db: SupportSQLiteDatabase) {

                db.execSQL(
                    """
                    ALTER TABLE pending_conversations
                    ADD COLUMN audioPath TEXT
                    """.trimIndent()
                )

                db.execSQL(
                    """
                    ALTER TABLE pending_conversations
                    ADD COLUMN status TEXT NOT NULL
                    DEFAULT 'WAITING_TRANSCRIPTION'
                    """.trimIndent()
                )
            }
        }

        private val MIGRATION_2_3 = object : Migration(2, 3) {
            override fun migrate(db: SupportSQLiteDatabase) {
                db.execSQL("DROP INDEX IF EXISTS index_conversations_recognitionId")
                db.execSQL("CREATE UNIQUE INDEX index_conversations_recognitionId ON conversations(recognitionId)")

                db.execSQL("DROP INDEX IF EXISTS index_person_history_recognitionId")
                db.execSQL("CREATE UNIQUE INDEX index_person_history_recognitionId ON person_history(recognitionId)")

                db.execSQL("DROP INDEX IF EXISTS index_pending_conversations_conversationId")
                db.execSQL("CREATE UNIQUE INDEX index_pending_conversations_conversationId ON pending_conversations(conversationId)")

                db.execSQL("CREATE UNIQUE INDEX index_emergency_contacts_priority ON emergency_contacts(priority)")
            }
        }
        /**
         * Opens the normal local database using the locally stored
         * SQLCipher database key.
         */
        fun getDatabase(context: Context): AppDatabase {

            return INSTANCE ?: synchronized(this) {
                INSTANCE ?: run {
                    val databaseKey =
                        DatabaseKeyManager(context).getDatabaseKey()

                    createDatabase(
                        context = context,
                        databaseKey = databaseKey
                    )
                }
            }
        }

        /**
         * Opens the database using an explicitly supplied key.
         *
         * Used when restoring a database from backup. The current instance
         * must be closed first, otherwise the supplied key would be silently
         * ignored.
         */
        fun getDatabaseWithKey(
            context: Context,
            databaseKey: ByteArray
        ): AppDatabase {

            return synchronized(this) {

                check(INSTANCE == null) {
                    "Close the existing database before opening it with another key."
                }

                createDatabase(
                    context = context,
                    databaseKey = databaseKey
                )
            }
        }

        private fun createDatabase(
            context: Context,
            databaseKey: ByteArray
        ): AppDatabase {

            require(databaseKey.size == 32) {
                "Database key must be 32 bytes."
            }

            // SQLCipher's native library must be loaded before first use.
            // Doing it here means it no longer depends on Application.onCreate.
            if (!nativeLibraryLoaded) {
                System.loadLibrary("sqlcipher")
                nativeLibraryLoaded = true
            }

            val factory =
                SupportOpenHelperFactory(databaseKey)

            val instance =
                Room.databaseBuilder(
                    context.applicationContext,
                    AppDatabase::class.java,
                    DATABASE_NAME
                )
                    .openHelperFactory(factory)
                    .addMigrations(MIGRATION_1_2, MIGRATION_2_3)
                    .build()

            INSTANCE = instance

            return instance
        }

        fun checkpointDatabase() {
            INSTANCE?.openHelper?.writableDatabase?.query(
                "PRAGMA wal_checkpoint(TRUNCATE)"
            )?.use {
                // Checkpoint executed.
            }
        }

        /**
         * Runs [block] while the database FILE is guaranteed to be complete
         * and unchanging, WITHOUT closing the database.
         *
         * Used for creating backups. Closing the database (the old approach)
         * left every repository, DAO and ViewModel pointing at a closed
         * instance.
         *
         * How it works:
         * 1. Checkpoint the WAL into the main database file.
         * 2. Take the write lock (a Room transaction), so no one can write.
         * 3. Verify the WAL is empty. If something wrote in between, release
         *    the lock and try again.
         *
         * Must not be called on the main thread.
         */
        fun <T> withQuiescedDatabase(
            context: Context,
            block: () -> T
        ): T {

            val database = getDatabase(context)
            val sqliteDatabase = database.openHelper.writableDatabase

            val walFile = File(
                requireNotNull(sqliteDatabase.path) {
                    "Database path is unavailable."
                } + "-wal"
            )

            repeat(QUIESCE_ATTEMPTS) {

                sqliteDatabase.query(
                    "PRAGMA wal_checkpoint(TRUNCATE)"
                ).use { cursor ->
                    cursor.moveToFirst()
                }

                val result = database.runInTransaction(
                    Callable<Quiesced<T>?> {

                        if (walFile.exists() && walFile.length() > 0L) {
                            null
                        } else {
                            Quiesced(block())
                        }
                    }
                )

                if (result != null) {
                    return result.value
                }
            }

            throw IllegalStateException(
                "The database kept changing while preparing the backup. Please try again."
            )
        }

        private class Quiesced<T>(val value: T)

        /**
         * Closes the database. Only use this when the app is about to
         * replace the database file (restore) or wipe all data, and
         * recreate its repositories / ViewModels afterwards (or restart).
         */
        fun closeDatabase() {
            synchronized(this) {
                INSTANCE?.close()
                INSTANCE = null
            }
        }
    }
}

package com.example.evoke.database

import androidx.room.Dao
import androidx.room.Delete
import androidx.room.Insert
import androidx.room.Query
import androidx.room.Update

// A bare coordinate returned by the bounding-box queries.
// The exact distance is computed in Kotlin (see LocationRepository),
// because SQLite on Android / SQLCipher is not guaranteed to have
// SIN, COS, ASIN, SQRT and POWER available.
data class GeoPoint(
    val id: Long,
    val latitude: Double,
    val longitude: Double
)

@Dao
interface LocationDao {

    @Insert
    suspend fun insert(location: Location): Long

    @Update
    suspend fun update(location: Location)

    @Delete
    suspend fun delete(location: Location)

    @Query("""
        SELECT *
        FROM locations
        ORDER BY locationId DESC
    """)
    suspend fun getAllLocations(): List<Location>

    @Query("""
        SELECT *
        FROM locations
        WHERE locationId = :locationId
        LIMIT 1
    """)
    suspend fun getById(locationId: Long): Location?

    // Saved locations inside a bounding box (uses the latitude/longitude indices).
    @Query("""
        SELECT locationId AS id, latitude, longitude
        FROM locations
        WHERE latitude BETWEEN :minLatitude AND :maxLatitude
          AND longitude BETWEEN :minLongitude AND :maxLongitude
    """)
    suspend fun getLocationsInBox(
        minLatitude: Double,
        maxLatitude: Double,
        minLongitude: Double,
        maxLongitude: Double
    ): List<GeoPoint>

    // Person-history points inside a bounding box.
    @Query("""
        SELECT historyId AS id, latitude, longitude
        FROM person_history
        WHERE latitude IS NOT NULL
          AND longitude IS NOT NULL
          AND latitude BETWEEN :minLatitude AND :maxLatitude
          AND longitude BETWEEN :minLongitude AND :maxLongitude
    """)
    suspend fun getHistoryPointsInBox(
        minLatitude: Double,
        maxLatitude: Double,
        minLongitude: Double,
        maxLongitude: Double
    ): List<GeoPoint>
}

package com.example.evoke.database

import androidx.room.ColumnInfo
import androidx.room.Entity
import androidx.room.ForeignKey
import androidx.room.Index
import androidx.room.PrimaryKey

// Represents the current stage of a pending conversation.
object PendingConversationStatus {

    // Audio has been recorded and is waiting for transcription.
    const val WAITING_TRANSCRIPTION = "WAITING_TRANSCRIPTION"

    // Transcription has been saved and is waiting for summarization.
    const val WAITING_SUMMARY = "WAITING_SUMMARY"
}

@Entity(
    tableName = "pending_conversations",
    foreignKeys = [
        ForeignKey(
            entity = Conversation::class,
            parentColumns = ["conversationId"],
            childColumns = ["conversationId"],
            onDelete = ForeignKey.CASCADE
        )
    ],
    indices = [ Index(value = ["conversationId"], unique = true) ]

)
data class PendingConversation(

    @PrimaryKey(autoGenerate = true)
    val pendingId: Long = 0,

    // Identifies the conversation this pending entry belongs to.
    val conversationId: Long,

    // Location of the temporarily stored filtered audio file.
    // Becomes null after the transcription has been safely saved
    // and the audio file has been deleted.
    val audioPath: String? = null,

    // Tracks which API needs to process this conversation next.
    @ColumnInfo(defaultValue = "'WAITING_TRANSCRIPTION'")
    val status: String = PendingConversationStatus.WAITING_TRANSCRIPTION
)

package com.example.evoke.database

import androidx.room.Dao
import androidx.room.Insert
import androidx.room.Query

@Dao
interface PendingConversationDao {

    // Creates a pending entry for a newly recorded conversation.
    @Insert
    suspend fun insert(
        pendingConversation: PendingConversation
    ): Long

    // Fetches ONLY the oldest pending conversation.
    // The worker will process this one before moving to the next.
    @Query("""
        SELECT *
        FROM pending_conversations
        ORDER BY pendingId ASC
        LIMIT 1
    """)
    suspend fun getOldestPending(): PendingConversation?

    // Counts all outstanding conversations.
    // The repository will use this to enforce the maximum of five.
    @Query("""
        SELECT COUNT(*)
        FROM pending_conversations
    """)
    suspend fun getPendingCount(): Int

    // Moves a pending conversation to the summarization stage.
    @Query("""
    UPDATE pending_conversations
    SET status = 'WAITING_SUMMARY'
    WHERE pendingId = :pendingId
""")
    suspend fun markWaitingForSummary(pendingId: Long): Int

    // Moves a pending conversation to the summarization stage by conversation id.
    @Query("""
    UPDATE pending_conversations
    SET status = 'WAITING_SUMMARY'
    WHERE conversationId = :conversationId
""")
    suspend fun markWaitingForSummaryByConversationId(
        conversationId: Long
    ): Int

    // Moves a pending conversation back to the transcription stage, if needed.
    @Query("""
    UPDATE pending_conversations
    SET status = 'WAITING_TRANSCRIPTION'
    WHERE pendingId = :pendingId
""")
    suspend fun markWaitingForTranscription(pendingId: Long): Int

    // Clears the audio path after the audio file has been deleted.
    @Query("""
        UPDATE pending_conversations
        SET audioPath = NULL
        WHERE pendingId = :pendingId
    """)
    suspend fun clearAudioPath(
        pendingId: Long
    ): Int

    // Deletes a completed pending entry.
    @Query("""
        DELETE FROM pending_conversations
        WHERE pendingId = :pendingId
    """)
    suspend fun deleteById(
        pendingId: Long
    ): Int

    // Deletes pending entries associated with a conversation.
    @Query("""
        DELETE FROM pending_conversations
        WHERE conversationId = :conversationId
    """)
    suspend fun deleteByConversationId(
        conversationId: Long
    )

    // Checks whether a conversation is still waiting for processing.
    @Query("""
        SELECT EXISTS(
            SELECT 1
            FROM pending_conversations
            WHERE conversationId = :conversationId
        )
    """)
    suspend fun isPending(
        conversationId: Long
    ): Boolean

    @Query("SELECT audioPath FROM pending_conversations WHERE audioPath IS NOT NULL")
    suspend fun getAllAudioPaths(): List<String>

    @Query("""
    SELECT pc.audioPath FROM pending_conversations pc
    INNER JOIN conversations c ON pc.conversationId = c.conversationId
    WHERE c.recognitionId = :recognitionId AND pc.audioPath IS NOT NULL
""")
    suspend fun getAudioPathsForRecognition(recognitionId: Long): List<String>

    @Query("""
    SELECT pc.audioPath FROM pending_conversations pc
    INNER JOIN conversations c ON pc.conversationId = c.conversationId
    INNER JOIN recognition_events re ON c.recognitionId = re.recognitionId
    WHERE re.personId = :personId AND pc.audioPath IS NOT NULL
""")
    suspend fun getAudioPathsForPerson(personId: Long): List<String>
}

        name: String,

package com.example.evoke.database.repository

import com.example.evoke.database.AppDatabase
import com.example.evoke.database.EmergencyContact
import androidx.room.withTransaction
import androidx.room.withTransaction

class EmergencyContactRepository(
    private val db: AppDatabase
) {

    private val dao = db.emergencyContactDao()

    /**
     * Replaces every saved emergency contact with [phoneNumbers], in order.
     *
     * Blank entries are skipped; the remaining numbers get priority 1..n in
     * the order given (first field = priority 1 = called / texted first).
     * At most 5 are kept. Used by the "Let us know you" / Profile form, whose
     * numbers EmergencyAlertService and LocationShareService read.
     */
    suspend fun replaceContacts(
        phoneNumbers: List<String>
    ) {
        val cleaned = phoneNumbers
            .map { it.trim() }
            .filter { it.isNotEmpty() }
            .take(5)

        require(cleaned.isNotEmpty()) {
            "At least 1 emergency contact is required."
        }

        db.withTransaction {
            dao.deleteAll()

            cleaned.forEachIndexed { index, phoneNo ->
                dao.insert(
                    EmergencyContact(
                        phoneNo = phoneNo,
                        priority = index + 1
                    )
                )
            }
        }
    }

    suspend fun addContact(
        contact: EmergencyContact
    ): Long {

        return db.withTransaction {

            // Check the maximum number of emergency contacts.
            val count = dao.getContactCount()

            if (count >= 5) {
                throw IllegalStateException(
                    "Maximum of 5 emergency contacts allowed."
                )
            }

            // Validate the priority.
            if (contact.priority !in 1..5) {
                throw IllegalArgumentException(
                    "Emergency contact priority must be between 1 and 5."
                )
            }

            // Make sure the priority isn't already assigned.
            if (dao.getByPriority(contact.priority) != null) {
                throw IllegalArgumentException(
                    "This priority is already assigned."
                )
            }

            // Insert only after all checks have passed.
            dao.insert(contact)
        }
    }

    suspend fun getContacts(): List<EmergencyContact> {
        return dao.getAllOrderedByPriority()
    }


    suspend fun deleteContact(
        contact: EmergencyContact
    ) {

        db.withTransaction {

            // At least one emergency contact must remain.
            val count = dao.getContactCount()

            if (count <= 1) {
                throw IllegalStateException(
                    "At least 1 emergency contact is required."
                )
            }

            dao.delete(contact)
        }
    }

    suspend fun getContactCount(): Int {
        return dao.getContactCount()
    }
}

package com.example.evoke.database.repository

import com.example.evoke.database.AppDatabase
import com.example.evoke.database.GeoPoint

import com.example.evoke.database.NearbyResult
import kotlin.math.asin
import kotlin.math.cos
import kotlin.math.max
import kotlin.math.min
import kotlin.math.pow
import kotlin.math.sin
import kotlin.math.sqrt

class LocationRepository(
    private val db: AppDatabase
) {

    private val locationDao = db.locationDao()

    companion object {
        private const val EARTH_RADIUS_KM = 6371.0
        private const val KM_PER_DEGREE = 111.32

        const val SOURCE_LOCATION = "LOCATION"
        const val SOURCE_PERSON_HISTORY = "PERSON_HISTORY"
    }

    suspend fun getAllLocations(): List<Location> {
        return locationDao.getAllLocations()
    }
    suspend fun addLocation(location: Location): Long {
        validateCoordinates(location.latitude, location.longitude)
        return locationDao.insert(location)
    }

    suspend fun updateLocation(location: Location) {
        validateCoordinates(location.latitude, location.longitude)
        locationDao.update(location)
    }
    suspend fun getLocation(
        locationId: Long
    ): Location? {
        return locationDao.getById(locationId)
    }

    suspend fun deleteLocation(
        location: Location
    ) {
        locationDao.delete(location)
    }

    /**
     * Finds saved locations and person-history points within [radiusKm]
     * of the given point, nearest first.
     *
     * SQL only narrows the search to a bounding box (using the indices).
     * The exact great-circle distance is computed here in Kotlin.
     *
     * Handles the antimeridian (longitude wrap) and the poles.
     *
     * If source == LOCATION        -> locationRepository.getLocation(id)
     * If source == PERSON_HISTORY  -> use the person history DAO with the id
     */
    suspend fun getNearbyResults(
        latitude: Double,
        longitude: Double,
        radiusKm: Double = 5.0
    ): List<NearbyResult> {

        require(latitude in -90.0..90.0) {
            "Latitude must be between -90 and 90."
        }

        require(longitude in -180.0..180.0) {
            "Longitude must be between -180 and 180."
        }

        require(radiusKm > 0.0) {
            "Radius must be greater than zero."
        }

        val latDelta = radiusKm / KM_PER_DEGREE

        val minLatitude = max(-90.0, latitude - latDelta)
        val maxLatitude = min(90.0, latitude + latDelta)

        val cosLatitude = cos(Math.toRadians(latitude))

        val reachesPole =
            latitude - latDelta <= -90.0 || latitude + latDelta >= 90.0

        // Near a pole every longitude is within range.
        val lonDelta =
            if (reachesPole || cosLatitude < 1e-6) {
                180.0
            } else {
                min(180.0, radiusKm / (KM_PER_DEGREE * cosLatitude))
            }

        val longitudeRanges = buildLongitudeRanges(longitude, lonDelta)

        val results = mutableListOf<NearbyResult>()

        for ((minLongitude, maxLongitude) in longitudeRanges) {

            locationDao.getLocationsInBox(
                minLatitude = minLatitude,
                maxLatitude = maxLatitude,
                minLongitude = minLongitude,
                maxLongitude = maxLongitude
            ).forEach { point ->
                addIfWithinRadius(
                    point = point,
                    source = SOURCE_LOCATION,
                    latitude = latitude,
                    longitude = longitude,
                    radiusKm = radiusKm,
                    results = results
                )
            }

            locationDao.getHistoryPointsInBox(
                minLatitude = minLatitude,
                maxLatitude = maxLatitude,
                minLongitude = minLongitude,
                maxLongitude = maxLongitude
            ).forEach { point ->
                addIfWithinRadius(
                    point = point,
                    source = SOURCE_PERSON_HISTORY,
                    latitude = latitude,
                    longitude = longitude,
                    radiusKm = radiusKm,
                    results = results
                )
            }
        }

        return results.sortedBy { it.distanceKm }
    }

    // Splits a longitude window into one or two ranges inside -180..180.
    private fun buildLongitudeRanges(
        longitude: Double,
        lonDelta: Double
    ): List<Pair<Double, Double>> {

        if (lonDelta >= 180.0) {
            return listOf(-180.0 to 180.0)
        }

        val minLongitude = longitude - lonDelta
        val maxLongitude = longitude + lonDelta

        return when {

            minLongitude < -180.0 -> listOf(
                (minLongitude + 360.0) to 180.0,
                -180.0 to maxLongitude
            )

            maxLongitude > 180.0 -> listOf(
                minLongitude to 180.0,
                -180.0 to (maxLongitude - 360.0)
            )

            else -> listOf(minLongitude to maxLongitude)
        }
    }

    private fun addIfWithinRadius(
        point: GeoPoint,
        source: String,
        latitude: Double,
        longitude: Double,
        radiusKm: Double,
        results: MutableList<NearbyResult>
    ) {
        val distance = haversineKm(
            latitude,
            longitude,
            point.latitude,
            point.longitude
        )

        if (distance <= radiusKm) {
            results.add(
                NearbyResult(
                    id = point.id,
                    latitude = point.latitude,
                    longitude = point.longitude,
                    distanceKm = distance,
                    source = source
                )
            )
        }
    }

    private fun haversineKm(
        lat1: Double,
        lon1: Double,
        lat2: Double,
        lon2: Double
    ): Double {

        val dLat = Math.toRadians(lat2 - lat1)
        val dLon = Math.toRadians(lon2 - lon1)

        val a =
            sin(dLat / 2).pow(2) +
                    cos(Math.toRadians(lat1)) *
                    cos(Math.toRadians(lat2)) *
                    sin(dLon / 2).pow(2)

        return 2.0 * EARTH_RADIUS_KM * asin(min(1.0, sqrt(a)))
    }
}

    private val voiceEmbeddingDao = db.voiceEmbeddingDao()
    private val recognitionDao = db.recognitionEventDao()
    private val historyDao = db.personHistoryDao()
    private val conversationDao = db.conversationDao()
    private val pendingDao = db.pendingConversationDao()

    companion object {
        const val MAX_PERSONS = 100
    }
    /**
     * Registers a person, their first embeddings and the first recognition
     * in ONE transaction.
     *
     * [profileImageProvider] is called inside the transaction, right after
     * the person row exists, with the new personId. It must save the profile
     * image and return its relative path. That way the image file name can
     * use the real personId, Person.profileImagePath is never left empty,
     * and if anything later in the transaction fails, the image is deleted.
     *
     * The repository takes ownership of [audioPath]'s file: it is deleted if
     * registration fails or the recording cannot be queued. Registration
     * itself never fails because the pending queue is full or GPS is missing.
     *
     *
     *  * A person must have a first recognition with a location or a queued
     *  * conversation. If neither is available (no GPS, voice disabled, or the
     *  * pending queue is full) registration fails and everything is rolled back.
     */
    suspend fun createPerson(
        person: Person,
        firstFaceEmbedding: FaceEmbedding,
        dateTime: Long,
        latitude: Double? = null,
        longitude: Double? = null,
        audioPath: String? = null,
        voiceEmbedding: VoiceEmbedding? = null,
        profileImageProvider: ((personId: Long) -> String)? = null
    ): Long {

        var savedImagePath: String? = null

        val saved = try {

            db.withTransaction {

                // Prevent registration when the maximum number of persons is reached.
                check(personDao.getPersonCount() < MAX_PERSONS) {
                    "The maximum limit of $MAX_PERSONS persons has been reached."
                }

                val personId = personDao.insert(
                    person.copy(
                        voiceEnabled = voiceEmbedding != null
                    )
                )

                if (profileImageProvider != null) {

                    val imagePath = profileImageProvider(personId)

                    savedImagePath = imagePath

                    personDao.updatePersonDetails(
                        personId = personId,
                        name = person.name,
                        relation = person.relation,
                        profileImagePath = imagePath
                    )
                }

                faceEmbeddingDao.insert(
                    firstFaceEmbedding.copy(
                        personId = personId
                    )
                )

                if (voiceEmbedding != null) {
                    voiceEmbeddingDao.insert(
                        voiceEmbedding.copy(
                            personId = personId
                        )
                    )
                }

                val recognition =
                    recognitionRepository.createRecognitionInternal(
                        personId = personId,
                        dateTime = dateTime,
                        latitude = latitude,
                        longitude = longitude,
                        audioPath = audioPath
                    )

                check(recognition.recognitionId != null) {
                    "A person needs a location or a recorded conversation."
                }

                personId to recognition.audioQueued
            }

        } catch (exception: Throwable) {

            // The transaction rolled back: remove the files it created.
            val imagePath = savedImagePath

            if (imagePath != null) {
                imageStorage.deleteProfileImage(imagePath)
            }

            audioStorage.deleteAudio(audioPath)

            throw exception
        }

        val (personId, audioQueued) = saved

        if (!audioQueued) {
            audioStorage.deleteAudio(audioPath)
        }

        return personId
    }

        embedding: FaceEmbedding
    ): Long {

        return db.withTransaction {

            // Check the current number of face embeddings for this person.
            val count =
                faceEmbeddingDao.getEmbeddingCount(embedding.personId)

            // If fewer than 3 exist, insert the new embedding.
            if (count < 3) {
                return@withTransaction faceEmbeddingDao.insert(embedding)
            }

            // Three already exist, so find the weakest one.
            val lowestQuality =
                faceEmbeddingDao.getLowestQualityEmbedding(
                    embedding.personId
                ) ?: throw IllegalStateException(
                    "Could not find the lowest-quality face embedding."
                )

            // Keep the existing embedding if the new one isn't better.
            if (embedding.quality <= lowestQuality.quality) {
                return@withTransaction lowestQuality.embeddingId
            }

            // Replace the weakest embedding with the better one.
            faceEmbeddingDao.delete(lowestQuality)

            faceEmbeddingDao.insert(embedding)
        }
    }

            val count =
                voiceEmbeddingDao.getEmbeddingCount(embedding.personId)

            // If fewer than 3 exist, insert the new embedding.
            if (count < 3) {

                val insertedId = voiceEmbeddingDao.insert(embedding)

                // A person with a voice sample can have conversations recorded.
                personDao.setVoiceEnabled(
                    personId = embedding.personId,
                    enabled = true
                )

                return@withTransaction insertedId
            }

            // Three already exist, so find the weakest one.
            val lowestQuality =
                voiceEmbeddingDao.getLowestQualityEmbedding(
                    embedding.personId
                ) ?: throw IllegalStateException(
                    "Could not find the lowest-quality voice embedding."
                )

            // Keep the existing embedding if the new one isn't better.
            if (embedding.quality <= lowestQuality.quality) {
                return@withTransaction lowestQuality.voiceEmbeddingId
            }

            // Replace the weakest embedding with the better one.
            voiceEmbeddingDao.delete(lowestQuality)

            voiceEmbeddingDao.insert(embedding)
        }
    }

    val audioQueued: Boolean
)

class RecognitionRepository(
    private val db: AppDatabase,
    private val audioStorage: AudioStorage
) {

    private val recognitionDao = db.recognitionEventDao()
    private val historyDao = db.personHistoryDao()
    private val conversationDao = db.conversationDao()
    private val pendingDao = db.pendingConversationDao()
    private val personDao = db.personDao()

    companion object {
        const val MAX_PENDING_CONVERSATIONS = 5

    }

    /**
     * Creates a recognition and its related records atomically.
     *
     * The repository takes ownership of [audioPath]'s file: it is deleted
     * if saving fails, or if the recording cannot be queued (voice disabled
     * or queue full). Check [isPendingQueueFull] beforehand if the UI should
     * warn the user.
     */
    suspend fun createRecognition(
        personId: Long,
        dateTime: Long,
        latitude: Double? = null,
        longitude: Double? = null,
        audioPath: String? = null
    ): Long {

        val hasLocation = latitude != null && longitude != null

        if (!hasLocation && audioPath.isNullOrBlank()) {
            throw IllegalArgumentException(
                "Recognition must contain either location or a conversation recording."
            )
        }

        val result = try {

            db.withTransaction {
                createRecognitionInternal(
                    personId = personId,
                    dateTime = dateTime,
                    latitude = latitude,
                    longitude = longitude,
                    audioPath = audioPath
                )
            }

        } catch (exception: Throwable) {

            // The transaction rolled back, so nothing references the file.
            audioStorage.deleteAudio(audioPath)

            throw exception
        }

        if (!result.audioQueued) {
            audioStorage.deleteAudio(audioPath)
        }

        return result.recognitionId
            ?: throw IllegalStateException(
                "Recognition was not saved: no location and the conversation could not be queued."
            )
    }

    /**
     * Internal version used when another repository already owns
     * the database transaction, such as PersonRepository.
     *
     * Does NOT delete audio files: the transaction owner does that after
     * it knows whether the transaction committed.
     *
     * A recognition event is always created. without a location or a queued conversation
     * it returns recognitionId = null, and the caller decides what to do. The allowEmpty
     * parameter can stay (default false) or you can delete it, since nothing will use it.
     */
    internal suspend fun createRecognitionInternal(
        personId: Long,
        dateTime: Long,
        latitude: Double? = null,
        longitude: Double? = null,
        audioPath: String? = null,
        allowEmpty: Boolean = false
    ): RecognitionSaveResult {

        validateCoordinates(latitude, longitude)

        var queueAudio = !audioPath.isNullOrBlank()

        if (queueAudio) {

            val voiceEnabled = personDao.isVoiceEnabled(personId)
                ?: throw IllegalArgumentException(
                    "Person does not exist."
                )

            // Enforced inside the transaction, so a sixth pending
            // conversation can never be inserted. When voice is off or the
            // queue is full the recording is skipped, not treated as an error.
            queueAudio =
                voiceEnabled &&
                        pendingDao.getPendingCount() < MAX_PENDING_CONVERSATIONS
        }

        // A recognition needs a location or a queued conversation.
        if (!allowEmpty && !(latitude != null && longitude != null) && !queueAudio) {
            return RecognitionSaveResult(recognitionId = null, audioQueued = false)
        }

        val recognitionId = recognitionDao.insert(
            RecognitionEvent(
                personId = personId,
                dateTime = dateTime
            )
        )

        // Save location history when both coordinates are available.
        if (latitude != null && longitude != null) {
            historyDao.insert(
                PersonHistory(
                    recognitionId = recognitionId,
                    latitude = latitude,
                    longitude = longitude
                )
            )
        }

        // Create a pending conversation when an audio recording is queued.
        if (queueAudio && audioPath != null) {

            val conversationId = conversationDao.insert(
                Conversation(
                    recognitionId = recognitionId,
                    summary = ""
                )
            )

            pendingDao.insert(
                PendingConversation(
                    conversationId = conversationId,
                    audioPath = audioPath
                    // Status defaults to WAITING_TRANSCRIPTION.
                )
            )
        }

        return RecognitionSaveResult(
            recognitionId = recognitionId,
            audioQueued = queueAudio
        )
    }

    /**
     * Saves the transcription and advances the pending conversation to the
     * summarization stage in ONE transaction. The same transaction clears
     * audioPath. The audio file is deleted after the transaction commits.
     *
     * The raw transcript is stored temporarily in Conversation.summary and is
     * replaced by the real summary in completeConversation(). The audio path is
     * cleared in the transaction and the file is deleted after commit; if that
     * delete fails, cleanupOrphanedAudio() removes it at the next start.
     */
    suspend fun saveTranscription(
        pendingId: Long,
        conversationId: Long,
        transcription: String
    ) {
        require(transcription.isNotBlank()) { "Transcription cannot be empty" }

        val audioPathToDelete = db.withTransaction {

            val pending = pendingDao.getOldestPending()
                ?: throw IllegalStateException("No pending conversation exists.")

            check(pending.pendingId == pendingId &&
                    pending.conversationId == conversationId) {
                "Pending conversation does not match."
            }

            check(pending.status == PendingConversationStatus.WAITING_TRANSCRIPTION) {
                "Pending conversation was not in WAITING_TRANSCRIPTION state."
            }

            // The raw transcript lives in summary until completeConversation().
            check(conversationDao.updateSummary(conversationId, transcription) == 1) {
                "Conversation does not exist."
            }

            check(pendingDao.markWaitingForSummary(pendingId) == 1) {
                "Unable to advance pending conversation to summarization."
            }

            check(pendingDao.clearAudioPath(pendingId) == 1) {
                "Unable to clear audio path."
            }

            pending.audioPath
        }

        audioStorage.deleteAudio(audioPathToDelete)
    }
    /**
     * Saves the final summary, discards the raw transcript and removes
     * the pending entry atomically.
     */
    suspend fun completeConversation(
        conversationId: Long,
        summary: String
    ) {

        require(summary.isNotBlank()) {
            "Summary cannot be empty."
        }

        val wordCount = summary.trim().split(Regex("\\s+")).size

        require(wordCount <= 250) {
            "Summary exceeds the 250-word limit."
        }

        db.withTransaction {

            val pending = pendingDao.getOldestPending()
                ?: throw IllegalStateException("No pending conversation exists.")

            check(pending.conversationId == conversationId) {
                "This conversation is not the next one in the queue."
            }

            check(pending.status == PendingConversationStatus.WAITING_SUMMARY) {
                "Conversation has not been transcribed yet."
            }

            check(conversationDao.updateSummary(conversationId, summary) == 1) {
                "Conversation does not exist."
            }
            check(pendingDao.deleteById(pending.pendingId) == 1) {
                "Unable to remove the completed pending entry."
            }

        }
    }


    /**
     * Drops a pending item immediately, for example when its audio file
     * is missing. The conversation row stays with an empty summary.
     */
    suspend fun discardPending(pendingId: Long) {

        val audioPathToDelete = db.withTransaction {
            val pending = pendingDao.getOldestPending()
                ?.takeIf { it.pendingId == pendingId }
                ?: return@withTransaction null

            pendingDao.deleteById(pendingId)
            pending.audioPath
        }

        audioStorage.deleteAudio(audioPathToDelete)
    }

    // Returns the total number of entries in the pending table,
    // regardless of whether they are waiting for transcription or summary.
    suspend fun getPendingConversationCount(): Int {
        return pendingDao.getPendingCount()
    }

    // True when no more recordings can be queued right now.
    suspend fun isPendingQueueFull(): Boolean {
        return pendingDao.getPendingCount() >= MAX_PENDING_CONVERSATIONS
    }

    // Deletes the recognition and any audio file still waiting for it.
    suspend fun deleteRecognition(
        recognitionId: Long
    ) {

        val audioPaths = db.withTransaction {

            val paths =
                pendingDao.getAudioPathsForRecognition(recognitionId)

            recognitionDao.deleteById(recognitionId)

            paths
        }

        audioPaths.forEach { path ->
            audioStorage.deleteAudio(path)
        }
    }

    // Returns only the oldest pending conversation.
    suspend fun getOldestPending(): PendingConversation? {
        return pendingDao.getOldestPending()
    }

    // Retrieves the conversation associated with a pending entry.
    // For a WAITING_SUMMARY entry, conversation.summary holds the raw transcript
    // that has to be summarized.
    suspend fun getConversation(
        conversationId: Long
    ): Conversation? {
        return conversationDao.getById(conversationId)
    }

    // True while a conversation is still waiting for transcription/summary.
    suspend fun isConversationPending(
        conversationId: Long
    ): Boolean {
        return pendingDao.isPending(conversationId)
    }

    // Clears the stored audio path after the audio file has been deleted.
    suspend fun clearPendingAudioPath(
        pendingId: Long
    ): Boolean {
        return pendingDao.clearAudioPath(pendingId) == 1
    }

    /**
     * Call once at app start, BEFORE any recording is saved.
     * Removes half-copied files and audio files that no database row
     * references any more.
     */
    suspend fun cleanupOrphanedAudio() {
        audioStorage.deleteTemporaryFiles()
        audioStorage.deleteUnreferencedAudio(
            pendingDao.getAllAudioPaths()
        )
    }
}

package com.example.evoke.database.repository

import androidx.room.withTransaction
import com.example.evoke.database.AppDatabase

class UserRepository(
    private val db: AppDatabase,
    private val audioStorage: AudioStorage,
    private val imageStorage: PersonImageStorage
) {

    private val userDao = db.userDao()
    private val emergencyContactDao = db.emergencyContactDao()
    private val userVoiceEmbeddingDao = db.userVoiceEmbeddingDao()

    private fun validateEmergencyContacts(
        emergencyContacts: List<EmergencyContact>
    ) {
        if (emergencyContacts.size !in 1..5) {
            throw IllegalArgumentException(
                "Emergency contacts must contain between 1 and 5 contacts."
            )
        }

        val priorities = emergencyContacts.map { it.priority }

        if (priorities.any { it !in 1..5 }) {
            throw IllegalArgumentException(
                "Emergency contact priority must be between 1 and 5."
            )
        }

        if (priorities.size != priorities.distinct().size) {
            throw IllegalArgumentException(
                "Emergency contact priorities must be unique."
            )
        }
    }

    suspend fun createUser(
        user: User,
        emergencyContacts: List<EmergencyContact>,
        voiceEmbeddings: List<UserVoiceEmbedding>
    ): Long {

        if (voiceEmbeddings.isEmpty()) {
            throw IllegalArgumentException(
                "At least one voice sample is required."
            )
        }

        if (voiceEmbeddings.size > 3) {
            throw IllegalArgumentException(
                "A maximum of 3 voice embeddings are allowed."
            )
        }

        validateEmergencyContacts(emergencyContacts)

        return db.withTransaction {

            check(userDao.getUser() == null) { "A user already exists." }

            val userId = userDao.insert(user)

            emergencyContacts.forEach { contact ->
                emergencyContactDao.insert(contact)
            }

            voiceEmbeddings.forEach { embedding ->
                userVoiceEmbeddingDao.insert(embedding)
            }

            userId
        }
    }
    suspend fun getUser(): UserWithDetails? {

        val user = userDao.getUser() ?: return null

        val emergencyContacts =
            emergencyContactDao.getAllOrderedByPriority()

        return UserWithDetails(
            user = user,
            emergencyContacts = emergencyContacts
        )
    }

    suspend fun updateUser(
        user: User,
        emergencyContacts: List<EmergencyContact>
    ) {
        validateEmergencyContacts(emergencyContacts)

        db.withTransaction {

            // Update user details
            check(userDao.update(user) == 1) { "User does not exist." }

            // Replace emergency contacts
            emergencyContactDao.deleteAll()

            emergencyContacts.forEach { contact ->
                emergencyContactDao.insert(contact)
            }
        }
    }
    /**
     * Wipes every table AND the files that belong to them (profile images
     * and conversation audio).
     *
     * clearAllTables() must not run inside withTransaction (it runs a
     * VACUUM) and must not run on the main thread, so it is called directly
     * on the IO dispatcher.
     *
     * The database key is NOT removed here. To also remove the key and the
     * database file, use AppDataManager.clearAllLocalData().
     */
    suspend fun deleteUserAndResetDatabase() {
        withContext(Dispatchers.IO) {
            db.clearAllTables()
            audioStorage.deleteAllAudio()
            imageStorage.deleteAllProfileImages()
        }
    }

    suspend fun addVoiceEmbedding(
        embedding: UserVoiceEmbedding
    ): Long {

        return db.withTransaction {

            // Retrieve the currently stored user voice embeddings.
            val embeddings =
                userVoiceEmbeddingDao.getAllEmbeddings()

            // If fewer than 3 exist, insert the new embedding.
            if (embeddings.size < 3) {
                return@withTransaction userVoiceEmbeddingDao.insert(embedding)
            }

            // Three already exist, so find the weakest one.
            val lowestQuality =
                userVoiceEmbeddingDao.getLowestQualityEmbedding()
                    ?: throw IllegalStateException(
                        "Could not find the lowest-quality user voice embedding."
                    )

            // Keep the existing embedding if the new one isn't better.
            if (embedding.quality <= lowestQuality.quality) {
                return@withTransaction lowestQuality.userVoiceEmbeddingId
            }

            // Replace the weakest embedding with the better one.
            userVoiceEmbeddingDao.delete(lowestQuality)

            userVoiceEmbeddingDao.insert(embedding)
        }
    }


//    I'd add a function to explicitly replace all user voice embeddings
//
//    This is useful if the user goes into settings and says something like "Re-register my voice":

    suspend fun replaceVoiceEmbeddings(
        embeddings: List<UserVoiceEmbedding>
    ) {
        if (embeddings.isEmpty()) {
            throw IllegalArgumentException(
                "At least one voice embedding is required."
            )
        }

        if (embeddings.size > 3) {
            throw IllegalArgumentException(
                "A maximum of 3 voice embeddings are allowed."
            )
        }

        db.withTransaction {

            userVoiceEmbeddingDao.deleteAll()

            embeddings.forEach { embedding ->
                userVoiceEmbeddingDao.insert(embedding)
            }
        }
    }
}

package com.example.evoke.backup

import android.content.Context
import java.io.File
import java.io.FileInputStream
import java.io.FileOutputStream
import java.util.zip.ZipEntry
import java.util.zip.ZipInputStream
import java.util.zip.ZipOutputStream

class BackupArchiveManager(
    private val context: Context
) {

    companion object {
        private const val DATABASE_FILE_NAME = "evoke_database"

        private const val DATABASE_ENTRY =
            "database/$DATABASE_FILE_NAME"

        private const val PEOPLE_DIRECTORY = "people"
        private const val PEOPLE_ENTRY = "people/"

        private const val AUDIO_DIRECTORY = "audio"
        private const val AUDIO_ENTRY = "audio/"

        private const val BUFFER_SIZE = 8192

        // Suffixes for the staged copy and the previous copy during a restore.
        private const val STAGED_SUFFIX = ".restore_new"
        private const val OLD_SUFFIX = ".restore_old"

        // UUID filenames with any optional alphanumeric extension.
        private val AUDIO_FILE_NAME_REGEX = Regex(
            "[0-9a-fA-F]{8}-[0-9a-fA-F]{4}-[0-9a-fA-F]{4}-[0-9a-fA-F]{4}-[0-9a-fA-F]{12}(\\.[A-Za-z0-9]+)?"
        )
    }

    private val databaseFile: File
        get() = context.getDatabasePath(DATABASE_FILE_NAME)

    // A stored audio file: UUID name, optional extension, never a temp file.
    private fun isStoredAudioName(name: String): Boolean {
        return AUDIO_FILE_NAME_REGEX.matches(name) &&
                !name.endsWith(".tmp", ignoreCase = true) &&
                !name.endsWith(".part", ignoreCase = true)
    }

    private val peopleDirectory: File
        get() = File(context.filesDir, PEOPLE_DIRECTORY)

    private val audioDirectory: File
        get() = File(context.filesDir, AUDIO_DIRECTORY)

    fun hasLocalBackupData(): Boolean {

        val databaseExists =
            databaseFile.exists() && databaseFile.length() > 0

        val peopleImagesExist =
            peopleDirectory.listFiles()?.any { it.isFile } == true

        val audioFilesExist =
            audioDirectory.listFiles()?.any { it.isFile } == true

        return databaseExists || peopleImagesExist || audioFilesExist
    }

    /**
     * Creates a ZIP archive containing:
     *
     * database/dementia_database
     *
     * people/
     *     person_1.jpg
     *     person_2.png
     *
     * audio/
     *     <UUID>.m4a
     *     <UUID>.wav
     *     <UUID>.mp3
     *
     * Audio extensions are preserved.
     * Temporary audio files are excluded.
     *
     * The resulting ZIP is NOT encrypted here.
     * BackupEncryptionManager handles encryption separately.
     */
    fun createBackupArchive(
        outputFile: File
    ) {

        require(databaseFile.exists()) {
            "Database file does not exist."
        }

        outputFile.parentFile?.mkdirs()

        ZipOutputStream(
            FileOutputStream(outputFile)
        ).use { zipOutputStream ->

            // Add database.
            addFileToZip(
                file = databaseFile,
                zipPath = DATABASE_ENTRY,
                zipOutputStream = zipOutputStream
            )

            // Add people directory.
            if (peopleDirectory.exists()) {

                zipOutputStream.putNextEntry(
                    ZipEntry(PEOPLE_ENTRY)
                )

                zipOutputStream.closeEntry()

                val peopleFiles = peopleDirectory.listFiles()
                    ?: throw IllegalStateException(
                        "Unable to read people directory."
                    )

                peopleFiles
                    .filter { it.isFile }
                    .forEach { imageFile ->

                        addFileToZip(
                            file = imageFile,
                            zipPath = "$PEOPLE_ENTRY${imageFile.name}",
                            zipOutputStream = zipOutputStream
                        )
                    }
            }

            // Add audio directory and valid stored audio files.
            if (audioDirectory.exists()) {

                zipOutputStream.putNextEntry(
                    ZipEntry(AUDIO_ENTRY)
                )

                zipOutputStream.closeEntry()

                val audioFiles = audioDirectory.listFiles()
                    ?: throw IllegalStateException(
                        "Unable to read audio directory."
                    )

                audioFiles
                    .filter {
                        it.isFile &&
                                isStoredAudioName(it.name)
                    }
                    .forEach { audioFile ->

                        addFileToZip(
                            file = audioFile,
                            zipPath = "$AUDIO_ENTRY${audioFile.name}",
                            zipOutputStream = zipOutputStream
                        )
                    }
            }
        }
    }

    /**
     * Restores a previously created backup archive.
     *
     * Extraction happens in a temporary directory first.
     * The live database and directories are not touched
     * until the archive has been successfully extracted.
     */
    fun extractBackupArchive(
        archiveFile: File
    ): File {

        require(archiveFile.exists()) {
            "Backup archive does not exist."
        }

        val temporaryDirectory =
            File(
                context.cacheDir,
                "backup_restore_${System.currentTimeMillis()}"
            )

        if (!temporaryDirectory.mkdirs()) {
            throw IllegalStateException(
                "Unable to create temporary restore directory."
            )
        }

        try {

            ZipInputStream(
                FileInputStream(archiveFile)
            ).use { zipInputStream ->

                while (true) {

                    val entry =
                        zipInputStream.nextEntry ?: break

                    validateZipEntry(entry)

                    // Resolve the entry to its canonical location.
                    val outputFile = File(
                        temporaryDirectory,
                        entry.name
                    ).canonicalFile

                    val canonicalTemporaryDirectory =
                        temporaryDirectory.canonicalFile

                    // Prevent ZIP path traversal.
                    require(
                        outputFile.path.startsWith(
                            canonicalTemporaryDirectory.path +
                                    File.separator
                        )
                    ) {
                        "Unsafe ZIP entry path detected."
                    }

                    if (entry.isDirectory) {

                        if (!outputFile.mkdirs() &&
                            !outputFile.exists()
                        ) {
                            throw IllegalStateException(
                                "Unable to create directory: ${entry.name}"
                            )
                        }

                    } else {

                        outputFile.parentFile?.mkdirs()

                        FileOutputStream(
                            outputFile
                        ).use { output ->

                            val buffer =
                                ByteArray(BUFFER_SIZE)

                            while (true) {

                                val bytesRead =
                                    zipInputStream.read(buffer)

                                if (bytesRead == -1) {
                                    break
                                }

                                output.write(
                                    buffer,
                                    0,
                                    bytesRead
                                )
                            }
                        }
                    }

                    zipInputStream.closeEntry()
                }
            }

            validateExtractedBackup(temporaryDirectory)

            return temporaryDirectory

        } catch (exception: Exception) {

            temporaryDirectory.deleteRecursively()

            throw exception
        }
    }

    /**
     * Swaps a successfully extracted backup into place so that it can be
     * undone.
     *
     * 1. The new database, people/ and audio/ are first copied next to the
     *    live locations ("*.restore_new"). If that fails, nothing live has
     *    been touched.
     * 2. The live data is then renamed to "*.restore_old" and the staged
     *    copies are renamed into place. Renames are cheap and stay on the
     *    same filesystem.
     * 3. The caller restores the database key, then calls
     *    [RestoreSwap.commit] (deletes the old data) or
     *    [RestoreSwap.rollback] (puts the old data back).
     *
     * The Room database must be closed before calling this.
     */
    fun swapInRestoredBackup(
        extractedDirectory: File
    ): RestoreSwap {

        val extractedDatabase =
            File(extractedDirectory, DATABASE_ENTRY)

        val extractedPeople =
            File(extractedDirectory, PEOPLE_DIRECTORY)

        val extractedAudio =
            File(extractedDirectory, AUDIO_DIRECTORY)

        require(extractedDatabase.exists()) {
            "Extracted backup database is missing."
        }

        val databaseDirectory =
            databaseFile.parentFile
                ?: throw IllegalStateException(
                    "Database directory is unavailable."
                )

        databaseDirectory.mkdirs()

        val sidecarSuffixes = listOf("-wal", "-shm", "-journal")

        val sidecars = sidecarSuffixes.map { suffix ->
            File(databaseFile.absolutePath + suffix)
        }

        val stagedDatabase =
            File(databaseDirectory, DATABASE_FILE_NAME + STAGED_SUFFIX)

        val stagedPeople =
            File(context.filesDir, PEOPLE_DIRECTORY + STAGED_SUFFIX)

        val stagedAudio =
            File(context.filesDir, AUDIO_DIRECTORY + STAGED_SUFFIX)

        val stagedCopies =
            listOf(stagedDatabase, stagedPeople, stagedAudio)

        fun oldCopyOf(file: File): File =
            File(file.parentFile, file.name + OLD_SUFFIX)

        val oldCopies =
            listOf(databaseFile, peopleDirectory, audioDirectory)
                .map { oldCopyOf(it) } +
                    sidecars.map { oldCopyOf(it) }

        // Leftovers from an earlier interrupted restore.
        (stagedCopies + oldCopies).forEach { it.deleteRecursively() }

        // ---- Step 1: stage. Nothing live is touched. ----
        try {

            extractedDatabase.copyTo(
                target = stagedDatabase,
                overwrite = true
            )

            if (extractedPeople.exists()) {
                check(
                    extractedPeople.copyRecursively(
                        target = stagedPeople,
                        overwrite = true
                    )
                ) {
                    "Unable to stage restored people images."
                }
            }

            if (extractedAudio.exists()) {
                check(
                    extractedAudio.copyRecursively(
                        target = stagedAudio,
                        overwrite = true
                    )
                ) {
                    "Unable to stage restored audio."
                }
            }

        } catch (exception: Exception) {

            stagedCopies.forEach { it.deleteRecursively() }

            throw exception
        }

        // ---- Step 2: swap, remembering how to undo each move. ----
        val undoActions = mutableListOf<() -> Unit>()

        fun move(from: File, to: File) {
            check(from.renameTo(to)) {
                "Unable to move ${from.name}."
            }
        }

        fun replace(live: File, staged: File) {

            val old = oldCopyOf(live)

            if (live.exists()) {
                move(live, old)
                undoActions.add { move(old, live) }
            }

            if (staged.exists()) {
                move(staged, live)
                undoActions.add { move(live, staged) }
            }
        }

        try {

            // Old SQLite sidecar files must never be paired with the new database.
            sidecars.forEach { sidecar ->
                if (sidecar.exists()) {
                    val old = oldCopyOf(sidecar)
                    move(sidecar, old)
                    undoActions.add { move(old, sidecar) }
                }
            }

            replace(databaseFile, stagedDatabase)
            replace(peopleDirectory, stagedPeople)
            replace(audioDirectory, stagedAudio)

        } catch (exception: Exception) {

            undoActions.asReversed().forEach { runCatching(it) }

            stagedCopies.forEach { it.deleteRecursively() }

            throw exception
        }

        return RestoreSwap(
            undoActions = undoActions,
            oldCopies = oldCopies,
            stagedCopies = stagedCopies
        )
    }

    /**
     * Removes a temporary extracted backup.
     */
    fun deleteTemporaryBackup(
        extractedDirectory: File
    ) {
        if (extractedDirectory.exists()) {
            extractedDirectory.deleteRecursively()
        }
    }

    private fun addFileToZip(
        file: File,
        zipPath: String,
        zipOutputStream: ZipOutputStream
    ) {

        zipOutputStream.putNextEntry(
            ZipEntry(zipPath)
        )

        FileInputStream(file).use { input ->

            val buffer =
                ByteArray(BUFFER_SIZE)

            while (true) {

                val bytesRead =
                    input.read(buffer)

                if (bytesRead == -1) {
                    break
                }

                zipOutputStream.write(
                    buffer,
                    0,
                    bytesRead
                )
            }
        }

        zipOutputStream.closeEntry()
    }

    /**
     * Prevents ZIP path traversal attacks.
     *
     * Only the database and direct files inside people/ and audio/
     * are accepted.
     */
    private fun validateZipEntry(
        entry: ZipEntry
    ) {

        val name = entry.name

        require(!name.startsWith("/")) {
            "Unsafe absolute ZIP path."
        }

        require(!name.contains("\\")) {
            "Unsafe ZIP path."
        }

        val normalizedParts = name.split("/")

        require(
            normalizedParts.none {
                it == ".." || it == "." || it.isEmpty()
            } || (entry.isDirectory &&
                    normalizedParts.lastOrNull() == "")
        ) {
            "Unsafe ZIP path traversal detected."
        }

        val isDatabaseEntry =
            name == DATABASE_ENTRY

        val isPeopleDirectory =
            entry.isDirectory && name == PEOPLE_ENTRY

        val isPeopleFile =
            !entry.isDirectory &&
                    name.startsWith(PEOPLE_ENTRY) &&
                    name.removePrefix(PEOPLE_ENTRY).isNotBlank() &&
                    !name.removePrefix(PEOPLE_ENTRY).contains("/")

        val isAudioDirectory =
            entry.isDirectory && name == AUDIO_ENTRY

        val audioFileName =
            name.removePrefix(AUDIO_ENTRY)

        val isAudioFile =
            !entry.isDirectory &&
                    name.startsWith(AUDIO_ENTRY) &&
                    isStoredAudioName(audioFileName)

        require(
            isDatabaseEntry ||
                    isPeopleDirectory ||
                    isPeopleFile ||
                    isAudioDirectory ||
                    isAudioFile
        ) {
            "Unexpected file in backup: $name"
        }
    }

    /**
     * Makes sure the archive contains the files we expect.
     */
    private fun validateExtractedBackup(
        directory: File
    ) {

        val extractedDatabase =
            File(
                directory,
                DATABASE_ENTRY
            )

        require(
            extractedDatabase.exists() &&
                    extractedDatabase.isFile &&
                    extractedDatabase.length() > 0
        ) {
            "Backup archive does not contain a valid database file."
        }

        val extractedPeopleDirectory =
            File(
                directory,
                PEOPLE_DIRECTORY
            )

        if (extractedPeopleDirectory.exists()) {

            require(extractedPeopleDirectory.isDirectory) {
                "Invalid people directory."
            }

            val peopleFiles =
                extractedPeopleDirectory.listFiles()
                    ?: throw IllegalStateException(
                        "Unable to read extracted people directory."
                    )

            peopleFiles.forEach { file ->
                require(file.isFile) {
                    "Invalid file inside people directory."
                }
            }
        }

        val extractedAudioDirectory = File(
            directory,
            AUDIO_DIRECTORY
        )

        if (extractedAudioDirectory.exists()) {

            require(extractedAudioDirectory.isDirectory) {
                "Invalid audio directory."
            }

            val audioFiles =
                extractedAudioDirectory.listFiles()
                    ?: throw IllegalStateException(
                        "Unable to read extracted audio directory."
                    )

            audioFiles.forEach { file ->

                require(file.isFile) {
                    "Invalid file inside audio directory."
                }

                require(
                    isStoredAudioName(file.name)
                ) {
                    "Unexpected file inside audio directory."
                }
            }
        }
    }
}


/**
 * Handle returned by BackupArchiveManager.swapInRestoredBackup().
 *
 * Call exactly one of [commit] or [rollback].
 */
class RestoreSwap internal constructor(
    private val undoActions: List<() -> Unit>,
    private val oldCopies: List<File>,
    private val stagedCopies: List<File>
) {

    // The restore succeeded: the previous data is no longer needed.
    fun commit() {
        oldCopies.forEach { it.deleteRecursively() }
    }

    // The restore failed: put the previous data back.
    fun rollback() {
        val failures = mutableListOf<Throwable>()

        undoActions.asReversed().forEach { undo ->
            try { undo() } catch (e: Throwable) { failures.add(e) }
        }
        stagedCopies.forEach { it.deleteRecursively() }

        if (failures.isNotEmpty()) {
            throw IllegalStateException(
                "Rollback incomplete: previous data may still be in the *.restore_old files.",
                failures.first()
            )
        }
    }
}

class BackupMetadataManager(
    private val context: Context

        private const val FORMAT_VERSION = 1
    }

    private val backupKeyManager =
        BackupKeyManager(context)

    private val databaseKeyManager =
        DatabaseKeyManager(context)

    private val passwordKeyManager =
        PasswordKeyManager()

    private val backupKeyWrapper =
        BackupKeyWrapper()

    private val backupDatabaseKeyManager =
        BackupDatabaseKeyManager()

    fun createMetadata(
        password: CharArray
    ): BackupMetadata {

        require(password.isNotEmpty()) {
            "Backup password cannot be empty."
        }

        var backupKey: ByteArray? = null
        var databaseKey: ByteArray? = null
        var kek: ByteArray? = null

        try {

            // Get the existing BackupKey.
            backupKey =
                backupKeyManager.getBackupKey()

            // Get the local SQLCipher DatabaseKey.
            databaseKey =
                databaseKeyManager.getDatabaseKeyForBackup()

            // Derive the KEK from the caregiver's password.
            val result =
                passwordKeyManager.createKey(
                    password
                )

            kek = result.first

            val kdfParameters =
                result.second

            // Wrap the BackupKey using the password-derived KEK.
            val wrapped =
                backupKeyWrapper.wrapBackupKey(
                    backupKey = backupKey,
                    kek = kek
                )

            // Encrypt the SQLCipher DatabaseKey using the BackupKey.
            val encryptedDatabaseKey =
                backupDatabaseKeyManager.encryptDatabaseKey(
                    databaseKey = databaseKey,
                    backupKey = backupKey
                )

            return BackupMetadata(

                formatVersion =
                    FORMAT_VERSION,

                kdfVersion =
                    kdfParameters.version,

                salt =
                    Base64.encodeToString(
                        kdfParameters.salt,
                        Base64.NO_WRAP
                    ),

                memoryKb =
                    kdfParameters.memoryKb,

                iterations =
                    kdfParameters.iterations,

                parallelism =
                    kdfParameters.parallelism,

                wrappedBackupKey =
                    wrapped.encryptedKey,

                backupKeyIv =
                    wrapped.iv,

                encryptedDatabaseKey =
                    encryptedDatabaseKey.encryptedKey,

                databaseKeyIv =
                    encryptedDatabaseKey.iv
            )

        } finally {
            backupKey?.fill(0)
            databaseKey?.fill(0)
            kek?.fill(0)
        }
    }
}

package com.example.evoke.backup

import android.content.Context
import android.util.Base64

class BackupPackageManager(
    private val context: Context
) {

    companion object {
        private const val BACKUP_PACKAGE_DIRECTORY =
            "backup_package"

        private const val ARCHIVE_FILE_NAME =
            "backup.zip"

        private const val ENCRYPTED_BACKUP_FILE_NAME =
            "backup.enc"

        private const val METADATA_FILE_NAME =
            "metadata.json"
    }

    private val archiveManager =
        BackupArchiveManager(context)

    private val encryptionManager =
        BackupEncryptionManager()

    private val metadataManager =
        BackupMetadataManager(context)

    private val backupKeyManager =
        BackupKeyManager(context)

    private val databaseKeyManager =
        DatabaseKeyManager(context)

    private val passwordKeyManager =
        PasswordKeyManager()

    private val backupKeyWrapper =
        BackupKeyWrapper()

    private val backupDatabaseKeyManager =
        BackupDatabaseKeyManager()

    /**
     * Creates a complete encrypted backup package.
     *
     * The returned directory contains only:
     *
     * metadata.json
     * backup.enc
     *
     * The unencrypted ZIP exists only temporarily and is deleted.
     */
    fun createBackup(
        password: CharArray
    ): File {

        require(password.isNotEmpty()) {
            "Backup password cannot be empty."
        }

        val packageDirectory =
            File(
                context.cacheDir,
                "${BACKUP_PACKAGE_DIRECTORY}_${System.currentTimeMillis()}"
            )

        packageDirectory.mkdirs()

        val temporaryArchive =
            File(
                packageDirectory,
                ARCHIVE_FILE_NAME
            )

        val encryptedBackup =
            File(
                packageDirectory,
                ENCRYPTED_BACKUP_FILE_NAME
            )

        val metadataFile =
            File(
                packageDirectory,
                METADATA_FILE_NAME
            )

        try {

            // ---------------------------------------------------------
            // 1. Make sure Room/SQLCipher is closed.
            //
            // This is important because the database may still have
            // changes in WAL files.
            // ---------------------------------------------------------

//            AppDatabase.closeDatabase()
//
//            // ---------------------------------------------------------
//            // 2. Create the unencrypted temporary ZIP.
//            //
//            // This contains:
//            // database/dementia_database
//            // people/*
//            // ---------------------------------------------------------
//
//            archiveManager.createBackupArchive(
//                temporaryArchive
//            )
// The database is NOT closed any more. Closing it left every
            // repository, DAO and ViewModel pointing at a closed instance.
            // withQuiescedDatabase creates the database file if needed,
            // checkpoints the WAL, holds the write lock and verifies the
            // WAL is empty while the archive is built.
            AppDatabase.withQuiescedDatabase(context) {
                archiveManager.createBackupArchive(
                    temporaryArchive
                )
            }
            // ---------------------------------------------------------
            // 3. Get the local BackupKey.
            // ---------------------------------------------------------

            val backupKey =
                backupKeyManager.getBackupKey()

            try {

                // -----------------------------------------------------
                // 4. Encrypt the ZIP using AES-256-GCM.
                // -----------------------------------------------------

                encryptionManager.encryptFile(
                    inputFile = temporaryArchive,
                    outputFile = encryptedBackup,
                    backupKey = backupKey
                )

                // -----------------------------------------------------
                // 5. Create cryptographic metadata.
                //
                // This contains:
                // - Argon2 parameters
                // - wrapped BackupKey
                // - encrypted DatabaseKey
                // - IVs
                // -----------------------------------------------------

                val metadata =
                    metadataManager.createMetadata(
                        password
                    )

                // -----------------------------------------------------
                // 6. Convert metadata to JSON.
                // -----------------------------------------------------

                val metadataJson =
                    JSONObject().apply {

                        put(
                            "formatVersion",
                            metadata.formatVersion
                        )

                        put(
                            "kdfVersion",
                            metadata.kdfVersion
                        )

                        put(
                            "salt",
                            metadata.salt
                        )

                        put(
                            "memoryKb",
                            metadata.memoryKb
                        )

                        put(
                            "iterations",
                            metadata.iterations
                        )

                        put(
                            "parallelism",
                            metadata.parallelism
                        )

                        put(
                            "wrappedBackupKey",
                            metadata.wrappedBackupKey
                        )

                        put(
                            "backupKeyIv",
                            metadata.backupKeyIv
                        )

                        put(
                            "encryptedDatabaseKey",
                            metadata.encryptedDatabaseKey
                        )

                        put(
                            "databaseKeyIv",
                            metadata.databaseKeyIv
                        )
                    }

                metadataFile.writeText(
                    metadataJson.toString()
                )

                // -----------------------------------------------------
                // 7. Delete the unencrypted ZIP.
                //
                // The final package must never contain it.
                // -----------------------------------------------------

                if (temporaryArchive.exists()) {
                    require(
                        temporaryArchive.delete()
                    ) {
                        "Unable to delete temporary backup archive."
                    }
                }

            } finally {

                backupKey.fill(0)
            }

            // ---------------------------------------------------------
            // 8. Verify final package.
            // ---------------------------------------------------------

            require(
                encryptedBackup.exists()
            ) {
                "Encrypted backup was not created."
            }

            require(
                metadataFile.exists()
            ) {
                "Backup metadata was not created."
            }

            require(
                !temporaryArchive.exists()
            ) {
                "Unencrypted backup archive still exists."
            }

            return packageDirectory

        } catch (exception: Exception) {

            // Never leave an unencrypted archive behind.
            temporaryArchive.delete()

            // Delete incomplete package.
            packageDirectory.deleteRecursively()

            throw exception

    private fun verifyRestoredDatabase(databaseKey: ByteArray) {
        try {
            val database = AppDatabase.getDatabaseWithKey(context, databaseKey)
            database.openHelper.writableDatabase
                .query("PRAGMA quick_check")
                .use { cursor ->
                    check(cursor.moveToFirst() && cursor.getString(0) == "ok") {

                    }
                }
        } finally {
            AppDatabase.closeDatabase()
        }
    }

    /**
     * Restores a complete backup package.
     *
     * The password is required to recover the BackupKey.
     */
    fun restoreBackup(
        packageDirectory: File,
        password: CharArray
    ) {

        require(password.isNotEmpty()) {
            "Backup password cannot be empty."
        }

        require(packageDirectory.exists()) {
            "Backup package does not exist."
        }

        val metadataFile =
            File(
                packageDirectory,
                METADATA_FILE_NAME
            )

        val encryptedBackup =
            File(
                packageDirectory,
                ENCRYPTED_BACKUP_FILE_NAME
            )

        val decryptedArchive =
            File(
                packageDirectory,
                "restored.zip"
            )

        var kek: ByteArray? = null
        var backupKey: ByteArray? = null
        var databaseKey: ByteArray? = null

        try {

            // ---------------------------------------------------------
            // 1. Read metadata.
            // ---------------------------------------------------------

            val metadata =
                readMetadata(
                    metadataFile
                )

            // ---------------------------------------------------------
            // 2. Recreate the KEK from the caregiver password.
            // ---------------------------------------------------------

            val kdfParameters =
                PasswordKeyManager.KdfParameters(
                    version = metadata.kdfVersion,

                    salt =
                        Base64.decode(
                            metadata.salt,
                            Base64.NO_WRAP
                        ),

                    memoryKb =
                        metadata.memoryKb,

                    iterations =
                        metadata.iterations,

                    parallelism =
                        metadata.parallelism
                )

            kek =
                passwordKeyManager.deriveKey(
                    password = password,
                    parameters = kdfParameters
                )

            // ---------------------------------------------------------
            // 3. Recover BackupKey.
            // ---------------------------------------------------------

            backupKey =
                backupKeyWrapper.unwrapBackupKey(
                    encryptedKey =
                        metadata.wrappedBackupKey,

                    iv =
                        metadata.backupKeyIv,

                    kek =
                        kek
                )

            // ---------------------------------------------------------
            // 4. Recover the SQLCipher DatabaseKey.
            // ---------------------------------------------------------

            databaseKey =
                backupDatabaseKeyManager.decryptDatabaseKey(
                    encryptedKey =
                        metadata.encryptedDatabaseKey,

                    iv =
                        metadata.databaseKeyIv,

                    backupKey =
                        backupKey
                )

            // ---------------------------------------------------------
            // 6. Decrypt backup.enc → restored.zip
            // ---------------------------------------------------------

            encryptionManager.decryptFile(
                encryptedFile = encryptedBackup,
                outputFile = decryptedArchive,
                backupKey = backupKey
            )

            // ---------------------------------------------------------
            // 7. Extract and validate the ZIP.
            //
            //    Nothing live is touched yet, so a wrong password or a
            //    corrupt backup can never damage the existing data.
            // ---------------------------------------------------------

            val extractedDirectory =
                archiveManager.extractBackupArchive(
                    decryptedArchive
                )

            // ---------------------------------------------------------
            // 8. Swap the restored data in, with rollback.
            //
            //    Order: files first (undoable), then the database key.
            //    If the key cannot be stored, the old files AND the old key
            //    are put back, so the app is never left with a database
            //    that its key cannot open.
            // ---------------------------------------------------------

            val restoredDatabaseKey =
                requireNotNull(databaseKey) {
                    "Database key was not recovered."
                }

            val previousDatabaseKey =
                runCatching { databaseKeyManager.getDatabaseKey() }.getOrNull()

            try {
                AppDatabase.closeDatabase()

                val swap = archiveManager.swapInRestoredBackup(extractedDirectory)

                try {
                    databaseKeyManager.restoreDatabaseKey(restoredDatabaseKey)
                    verifyRestoredDatabase(restoredDatabaseKey)
                } catch (exception: Exception) {
                    // Key first, then files, and never let one failure hide the other.
                    previousDatabaseKey?.let { previous ->
                        runCatching { databaseKeyManager.restoreDatabaseKey(previous) }
                            .onFailure { exception.addSuppressed(it) }
                    }
                    runCatching { swap.rollback() }
                        .onFailure { exception.addSuppressed(it) }
                    throw exception
                }

                // Only reached when the restored DB opened and passed its check.
                swap.commit()

            } finally {
                previousDatabaseKey?.fill(0)
                archiveManager.deleteTemporaryBackup(extractedDirectory)
            }

            // After a successful restore every repository / ViewModel
            // still references the old database instance. Restart the
            // app (or recreate AppViewModelFactory and all ViewModels).

            // ---------------------------------------------------------
            // 10. Delete decrypted temporary ZIP.
            // ---------------------------------------------------------

            if (decryptedArchive.exists()) {
                decryptedArchive.delete()
            }

        } catch (exception: Exception) {

            decryptedArchive.delete()

            throw exception

        } finally {
            kek?.fill(0)
            backupKey?.fill(0)
            databaseKey?.fill(0)
        }
    }
    fun getEncryptedBackupFile(
        packageDirectory: File
    ): File {

        return File(
            packageDirectory,
            ENCRYPTED_BACKUP_FILE_NAME
        )
    }

    fun getMetadataFile(

    ): File {

        return File(
            packageDirectory,
            METADATA_FILE_NAME
        )
    }
    fun hasLocalBackupData(): Boolean {
        return archiveManager.hasLocalBackupData()
    }
}

class GoogleDriveBackupManager {

    companion object {

        private const val DRIVE_FILES_URL =
            "https://www.googleapis.com/drive/v3/files"

        private const val DRIVE_UPLOAD_URL =
            "https://www.googleapis.com/upload/drive/v3/files"

        private const val APP_DATA_FOLDER =
            "appDataFolder"

        private const val BACKUP_FILE_NAME =
            "evoke_backup.enc"

        private const val METADATA_FILE_NAME =
            "evoke_backup_metadata.json"

        // Names used LOCALLY after download. They must match
        // BackupPackageManager (backup.enc / metadata.json), otherwise
        // restoreBackup() cannot find the files.
        private const val LOCAL_BACKUP_FILE_NAME =
            "backup.enc"

        private const val LOCAL_METADATA_FILE_NAME =
            "metadata.json"
    }

    /**
     * Uploads or replaces the encrypted backup and its metadata
     * inside Google's private appDataFolder.
     */
    suspend fun uploadBackup(
        accessToken: String,
        backupFile: File,
        metadataFile: File
    ): DriveBackupUploadResult =
        withContext(Dispatchers.IO) {

            require(accessToken.isNotBlank()) {
                "Google Drive access token is missing."
            }

            require(backupFile.exists()) {
                "Encrypted backup file does not exist."
            }

            require(metadataFile.exists()) {
                "Backup metadata file does not exist."
            }

            try {

                val backupFileId =
                    findFile(
                        accessToken = accessToken,
                        fileName = BACKUP_FILE_NAME
                    )

                val metadataFileId =
                    findFile(
                        accessToken = accessToken,
                        fileName = METADATA_FILE_NAME
                    )

                val uploadedBackupId =
                    if (backupFileId != null) {

                        updateFile(
                            accessToken = accessToken,
                            fileId = backupFileId,
                            file = backupFile,
                            mimeType = "application/octet-stream"
                        )

                    } else {

                        createFile(
                            accessToken = accessToken,
                            fileName = BACKUP_FILE_NAME,
                            file = backupFile,
                            mimeType = "application/octet-stream"
                        )
                    }

                val uploadedMetadataId =
                    if (metadataFileId != null) {

                        updateFile(
                            accessToken = accessToken,
                            fileId = metadataFileId,
                            file = metadataFile,
                            mimeType = "application/json"
                        )

                    } else {

                        createFile(
                            accessToken = accessToken,
                            fileName = METADATA_FILE_NAME,
                            file = metadataFile,
                            mimeType = "application/json"
                        )
                    }

                DriveBackupUploadResult.Success(
                    backupFileId = uploadedBackupId,
                    metadataFileId = uploadedMetadataId
                )

            } catch (e: Exception) {

                DriveBackupUploadResult.Failed(
                    e.message
                        ?: "Google Drive backup upload failed."
                )
            }
        }

    /**

    /**
     * Finds a file belonging to this app's private appDataFolder.
     */
    private fun findFile(
        accessToken: String,
        fileName: String
    ): String? {

        val encodedName =
            fileName
                .replace("\\", "\\\\")
                .replace("'", "\\'")

        val query =
            "'$APP_DATA_FOLDER' in parents " +
                    "and name = '$encodedName' " +
                    "and trashed = false"

        val encodedQuery =
            URLEncoder.encode(
                query,
                "UTF-8"
            )

        val url =
            URL(
                "$DRIVE_FILES_URL" +
                        "?spaces=appDataFolder" +
                        "&q=$encodedQuery" +
                        "&fields=files(id,name)"
            )

        val connection =
            url.openConnection() as HttpURLConnection

        try {

            connection.requestMethod = "GET"

            connection.setRequestProperty(
                "Authorization",
                "Bearer $accessToken"
            )

            connection.setRequestProperty(
                "Accept",
                "application/json"
            )

            val responseCode =
                connection.responseCode

            if (responseCode !in 200..299) {

                throw Exception(
                    "Google Drive file search failed. " +
                            "HTTP $responseCode"
                )
            }

            val response =
                connection.inputStream
                    .bufferedReader()
                    .use {
                        it.readText()
                    }

            val json =
                JSONObject(response)

            val files =
                json.optJSONArray("files")

            if (
                files == null ||
                files.length() == 0
            ) {
                return null
            }

            return files
                .getJSONObject(0)
                .getString("id")

        } finally {

            connection.disconnect()
        }
    }

    /**
     * Downloads a single file from Google Drive.
     */
    private fun downloadFile(
        accessToken: String,
        fileId: String,
        outputFile: File
    ) {

        val url =
            URL(
                "$DRIVE_FILES_URL/$fileId?alt=media"
            )

        val connection =
            url.openConnection() as HttpURLConnection

        try {

            connection.requestMethod = "GET"

            connection.setRequestProperty(
                "Authorization",
                "Bearer $accessToken"
            )

            connection.setRequestProperty(
                "Accept",
                "*/*"
            )

            val responseCode =
                connection.responseCode

            if (responseCode !in 200..299) {

                val errorBody =
                    try {
                        connection.errorStream
                            ?.bufferedReader()
                            ?.use {
                                it.readText()
                            }
                    } catch (_: Exception) {
                        null
                    }

                throw Exception(
                    "Google Drive backup download failed. " +
                            "HTTP $responseCode" +
                            if (!errorBody.isNullOrBlank()) {
                                "\n$errorBody"
                            } else {
                                ""
                            }
                )
            }

            val temporaryFile =
                File(
                    outputFile.parentFile,
                    "${outputFile.name}.tmp"
                )

            try {

                connection.inputStream.use { input ->

                    temporaryFile.outputStream().use { output ->

                        input.copyTo(
                            output,
                            bufferSize = 8192
                        )
                    }
                }

                require(
                    temporaryFile.renameTo(outputFile)
                ) {
                    "Unable to save downloaded backup file."
                }

            } catch (e: Exception) {

                temporaryFile.delete()
                throw e
            }

        } finally {

            connection.disconnect()
        }
    }

    /**
     * Creates a new file inside appDataFolder.
     */
    private fun createFile(
        accessToken: String,
        fileName: String,
        file: File,
        mimeType: String
    ): String {

        val url =
            URL(
                "$DRIVE_UPLOAD_URL" +
                        "?uploadType=multipart" +
                        "&fields=id"
            )

        val boundary =
            "EvokeBackupBoundary"

        val connection =
            url.openConnection() as HttpURLConnection

        try {

            connection.requestMethod = "POST"

            connection.doOutput = true

            // Stream the upload instead of buffering the whole body
            // (backups can contain audio) in memory.
            connection.setChunkedStreamingMode(64 * 1024)

            connection.setRequestProperty(
                "Authorization",
                "Bearer $accessToken"
            )

            connection.setRequestProperty(
                "Content-Type",
                "multipart/related; boundary=$boundary"
            )

            val metadata =
                JSONObject().apply {

                    put(
                        "name",
                        fileName
                    )

                    put(
                        "parents",
                        org.json.JSONArray().apply {
                            put(APP_DATA_FOLDER)
                        }
                    )
                }

            connection.outputStream.use { output ->

                writeMultipart(
                    output = output,
                    boundary = boundary,
                    metadata = metadata.toString(),
                    file = file,
                    mimeType = mimeType
                )
            }

            val responseCode =
                connection.responseCode

            if (responseCode !in 200..299) {

                throw Exception(
                    "Google Drive upload failed. " +
                            "HTTP $responseCode"
                )
            }

            val response =
                connection.inputStream
                    .bufferedReader()
                    .use {
                        it.readText()
                    }

            return JSONObject(response)
                .getString("id")

        } finally {

            connection.disconnect()
        }
    }

     * Replaces the content of an existing Drive file.
     */
    private fun updateFile(
        accessToken: String,
        fileId: String,
        file: File,
        mimeType: String
    ): String {

        val url =
            URL(
                "$DRIVE_UPLOAD_URL/$fileId" +
                        "?uploadType=media" +
                        "&fields=id"
            )

        val connection =
            url.openConnection() as HttpURLConnection

        try {

            connection.requestMethod = "PATCH"

            connection.doOutput = true

            connection.setRequestProperty(
                "Authorization",
                "Bearer $accessToken"
            )

            connection.setRequestProperty(
                "Content-Type",
                mimeType
            )

            // Content-Length set through setRequestProperty is ignored by
            // HttpURLConnection, which then buffers the whole body in memory.
            // A fixed-length streaming mode streams the file instead.
            connection.setFixedLengthStreamingMode(file.length())

            file.inputStream().use { input ->
                connection.outputStream.use { output ->

                    input.copyTo(
                        output,
                        bufferSize = 8192
                    )
                }
            }

            val responseCode =
                connection.responseCode

            if (responseCode !in 200..299) {

                throw Exception(
                    "Google Drive update failed. " +
                            "HTTP $responseCode"
                )
            }

            val response =
                connection.inputStream
                    .bufferedReader()
                    .use {
                        it.readText()
                    }

            return JSONObject(response)
                .getString("id")

        } finally {

            connection.disconnect()
        }
    }

    private fun writeMultipart(
        output: java.io.OutputStream,
        boundary: String,
        metadata: String,
        file: File,
        mimeType: String
    ) {

        val lineBreak =
            "\r\n"

        output.write(
            "--$boundary$lineBreak"
                .toByteArray()
        )

        output.write(
            "Content-Type: application/json; charset=UTF-8$lineBreak"
                .toByteArray()
        )

        output.write(
            lineBreak.toByteArray()
        )

        output.write(
            metadata.toByteArray()
        )

        output.write(
            lineBreak.toByteArray()
        )

        output.write(
            "--$boundary$lineBreak"
                .toByteArray()
        )

        output.write(
            "Content-Type: $mimeType$lineBreak"
                .toByteArray()
        )

        output.write(
            lineBreak.toByteArray()
        )

        file.inputStream().use { input ->

            input.copyTo(
                output,
                bufferSize = 8192
            )
        }

        output.write(
            lineBreak.toByteArray()
        )

        output.write(
            "--$boundary--$lineBreak"
                .toByteArray()
        )

        output.flush()
    }

    /**

     * already exists in Google's private appDataFolder.

    suspend fun hasExistingBackup(

    ): Boolean = withContext(Dispatchers.IO) {

    }

class GoogleDriveManager {

    companion object {
        private const val DRIVE_APP_DATA_SCOPE =
            "https://www.googleapis.com/auth/drive.appdata"
    }

    private val driveScope =
        Scope(DRIVE_APP_DATA_SCOPE)

    suspend fun requestDriveAccess(
        activity: Activity
    ): DriveAuthorizationResult {

        // Check internet before requesting Drive authorization.
        if (!isInternetAvailable(activity)) {
            throw NoInternetConnectionException()
        }

        val request =
            AuthorizationRequest.Builder()
                .setRequestedScopes(
                    listOf(driveScope)
                )
                .build()

        return try {

            val authorizationResult =
                Identity
                    .getAuthorizationClient(activity)
                    .authorize(request)
                    .await()

            if (authorizationResult.hasResolution()) {

                DriveAuthorizationResult.NeedsResolution(
                    authorizationResult.pendingIntent
                )

            } else {

                DriveAuthorizationResult.Authorized(
                    authorizationResult.accessToken
                )
            }

        } catch (e: CancellationException) {

            throw e

        } catch (e: Exception) {

            if (!isInternetAvailable(activity)) {
                throw NoInternetConnectionException()
            }

            throw e
        }
    }

    fun getAuthorizationResultFromIntent(
        activity: Activity,
        data: Intent?
    ): DriveAuthorizationResult {

        if (data == null) {
            return DriveAuthorizationResult.Failed(
                "Authorization was cancelled."
            )
        }

        return try {

            val result =
                Identity
                    .getAuthorizationClient(activity)
                    .getAuthorizationResultFromIntent(data)

            DriveAuthorizationResult.Authorized(
                result.accessToken
            )

        } catch (e: Exception) {

            if (!isInternetAvailable(activity)) {

                DriveAuthorizationResult.Failed(
                    "No internet connection. Please connect to the internet and try again."
                )

            } else {

                DriveAuthorizationResult.Failed(
                    "Google Drive authorization failed. Please try again."
                )
            }
        }
    }

    private fun isInternetAvailable(
        activity: Activity
    ): Boolean {

        val connectivityManager =
            activity.getSystemService(
                Activity.CONNECTIVITY_SERVICE
            ) as ConnectivityManager

        val network =
            connectivityManager.activeNetwork
                ?: return false

        val capabilities =
            connectivityManager.getNetworkCapabilities(network)
                ?: return false

        return capabilities.hasCapability(
            NetworkCapabilities.NET_CAPABILITY_INTERNET
        ) && capabilities.hasCapability(
            NetworkCapabilities.NET_CAPABILITY_VALIDATED
        )
    }
}

            secretKey,
            spec

        return EncryptedDatabaseKey(
            encryptedKey = Base64.encodeToString(
                encryptedKey,
                Base64.NO_WRAP
            ),
            iv = Base64.encodeToString(
                iv,
                Base64.NO_WRAP
            )
        )
    }

     */

package com.example.evoke.security

import java.io.File
import java.io.FileInputStream
import java.io.FileOutputStream
import java.security.SecureRandom
import javax.crypto.Cipher
import javax.crypto.spec.GCMParameterSpec
import javax.crypto.spec.SecretKeySpec

class BackupEncryptionManager {

    companion object {
        private const val AES_MODE = "AES/GCM/NoPadding"
        private const val AES_KEY_SIZE = 32
        private const val GCM_TAG_LENGTH = 128
        private const val IV_SIZE = 12
    }

    /**
     * Encrypts [inputFile] using AES-256-GCM and writes the result
     * to [outputFile].
     *
     * File format:
     *
     * [12-byte IV][encrypted data + GCM authentication tag]
     */
    fun encryptFile(
        inputFile: File,
        outputFile: File,
        backupKey: ByteArray
    ) {
        require(backupKey.size == AES_KEY_SIZE) {
            "Backup key must be 32 bytes."
        }

        require(inputFile.exists()) {
            "Input file does not exist: ${inputFile.absolutePath}"
        }

        val iv = ByteArray(IV_SIZE)
        SecureRandom().nextBytes(iv)

        val secretKey =
            SecretKeySpec(
                backupKey,
                "AES"
            )

        val cipher =
            Cipher.getInstance(AES_MODE)

        cipher.init(
            Cipher.ENCRYPT_MODE,
            secretKey,
            GCMParameterSpec(
                GCM_TAG_LENGTH,
                iv
            )
        )

        FileInputStream(inputFile).use { input ->
            FileOutputStream(outputFile).use { output ->

                // Store IV at the beginning of the encrypted file.
                output.write(iv)

                val buffer = ByteArray(8192)

                while (true) {

                    val bytesRead =
                        input.read(buffer)

                    if (bytesRead == -1) {
                        break
                    }

                    val encryptedChunk =
                        cipher.update(
                            buffer,
                            0,
                            bytesRead
                        )

                    if (encryptedChunk != null) {
                        output.write(encryptedChunk)
                    }
                }

                val finalBytes =
                    cipher.doFinal()

                output.write(finalBytes)
            }
        }
    }

    /**
     * Decrypts an AES-256-GCM encrypted backup file.
     *
     * The first 12 bytes of the file contain the IV.
     */
    fun decryptFile(
        encryptedFile: File,
        outputFile: File,
        backupKey: ByteArray
    ) {
        require(backupKey.size == AES_KEY_SIZE) {
            "Backup key must be 32 bytes."
        }

        require(encryptedFile.exists()) {
            "Encrypted file does not exist: ${encryptedFile.absolutePath}"
        }

        val secretKey =
            SecretKeySpec(
                backupKey,
                "AES"
            )

        FileInputStream(encryptedFile).use { input ->

            val iv = ByteArray(IV_SIZE)

            val ivBytesRead =
                input.read(iv)

            require(ivBytesRead == IV_SIZE) {
                "Invalid encrypted backup: IV is missing or incomplete."
            }

            val cipher =
                Cipher.getInstance(AES_MODE)

            cipher.init(
                Cipher.DECRYPT_MODE,
                secretKey,
                GCMParameterSpec(
                    GCM_TAG_LENGTH,
                    iv
                )
            )

            val temporaryOutputFile =
                File(
                    outputFile.parentFile,
                    "${outputFile.name}.tmp"
                )

            try {
                FileOutputStream(
                    temporaryOutputFile
                ).use { output ->

                    val buffer =
                        ByteArray(8192)

                    while (true) {
                        val bytesRead =
                            input.read(buffer)

                        if (bytesRead == -1) {
                            break
                        }

                        val decryptedChunk =
                            cipher.update(
                                buffer,
                                0,
                                bytesRead
                            )

                        if (decryptedChunk != null) {
                            output.write(
                                decryptedChunk
                            )
                        }
                    }

                    // Authentication is completed here.
                    val finalBytes =
                        cipher.doFinal()

                    output.write(finalBytes)

                    output.flush()
                }

                // Only expose the decrypted file after
                // successful GCM authentication.
                if (outputFile.exists()) {
                    outputFile.delete()
                }

                require(
                    temporaryOutputFile.renameTo(
                        outputFile
                    )
                ) {
                    "Unable to finalize decrypted backup."
                }

            } catch (exception: Exception) {
                temporaryOutputFile.delete()
                throw exception
            }
        }
    }
}

package com.example.evoke.security

import android.content.Context
import android.util.Base64
import android.security.keystore.KeyGenParameterSpec
import android.security.keystore.KeyProperties
import java.security.KeyStore
import java.security.SecureRandom
import javax.crypto.Cipher
import javax.crypto.KeyGenerator
import javax.crypto.SecretKey
import javax.crypto.spec.GCMParameterSpec

class BackupKeyManager(
    private val context: Context
) {

    companion object {
        private const val KEYSTORE_PROVIDER = "AndroidKeyStore"
        private const val MASTER_KEY_ALIAS = "dementia_backup_master_key"

        private const val PREFS_NAME = "backup_security"
        private const val ENCRYPTED_BACKUP_KEY = "encrypted_backup_key"
        private const val BACKUP_KEY_IV = "backup_key_iv"

        private const val AES_MODE = "AES/GCM/NoPadding"
        private const val GCM_TAG_LENGTH = 128

        private const val BACKUP_KEY_SIZE = 32
    }

    private val preferences =
        context.getSharedPreferences(
            PREFS_NAME,
            Context.MODE_PRIVATE
        )

    /**
     * Returns the application's backup-encryption key.
     *
     * The key itself is random and is never stored as plaintext
     * in SharedPreferences.
     */
    fun getBackupKey(): ByteArray {

        val encryptedKey =
            preferences.getString(
                ENCRYPTED_BACKUP_KEY,
                null
            )

        val iv =
            preferences.getString(
                BACKUP_KEY_IV,
                null
            )

        return if (
            encryptedKey == null ||
            iv == null
        ) {

            val backupKey =
                generateBackupKey()

            saveBackupKey(backupKey)

            backupKey

        } else {

            decryptBackupKey(
                encryptedKey = Base64.decode(
                    encryptedKey,
                    Base64.DEFAULT
                ),
                iv = Base64.decode(
                    iv,
                    Base64.DEFAULT
                )
            )
        }
    }

    /**
     * Generates a cryptographically secure 256-bit backup key.
     */
    private fun generateBackupKey(): ByteArray {

        val key =
            ByteArray(BACKUP_KEY_SIZE)

        SecureRandom().nextBytes(key)

        return key
    }

    /**
     * Encrypts the backup key using an Android Keystore master key
     * before storing it in SharedPreferences.
     */
    private fun saveBackupKey(
        backupKey: ByteArray
    ) {

        val cipher =
            Cipher.getInstance(AES_MODE)

        cipher.init(
            Cipher.ENCRYPT_MODE,
            getOrCreateMasterKey()
        )

        val encrypted =
            cipher.doFinal(backupKey)

        val iv =
            cipher.iv

        preferences.edit()
            .putString(
                ENCRYPTED_BACKUP_KEY,
                Base64.encodeToString(
                    encrypted,
                    Base64.NO_WRAP
                )
            )
            .putString(
                BACKUP_KEY_IV,
                Base64.encodeToString(
                    iv,
                    Base64.NO_WRAP
                )
            )
            .apply()
    }

    /**
     * Decrypts the locally stored backup key.
     */
    private fun decryptBackupKey(
        encryptedKey: ByteArray,
        iv: ByteArray
    ): ByteArray {

        val cipher =
            Cipher.getInstance(AES_MODE)

        val spec =
            GCMParameterSpec(
                GCM_TAG_LENGTH,
                iv
            )

        cipher.init(
            Cipher.DECRYPT_MODE,
            getOrCreateMasterKey(),
            spec
        )

        return cipher.doFinal(encryptedKey)
    }

    /**
     * Gets the Android Keystore master key or creates it
     * if this is the first time backup security is initialized.
     */
    private fun getOrCreateMasterKey(): SecretKey {

        val keyStore =
            KeyStore.getInstance(
                KEYSTORE_PROVIDER
            ).apply {
                load(null)
            }

        val existingKey =
            keyStore.getKey(
                MASTER_KEY_ALIAS,
                null
            ) as? SecretKey

        if (existingKey != null) {
            return existingKey
        }

        val keyGenerator =
            KeyGenerator.getInstance(
                KeyProperties.KEY_ALGORITHM_AES,
                KEYSTORE_PROVIDER
            )

        keyGenerator.init(
            KeyGenParameterSpec.Builder(
                MASTER_KEY_ALIAS,
                KeyProperties.PURPOSE_ENCRYPT or
                        KeyProperties.PURPOSE_DECRYPT
            )
                .setBlockModes(
                    KeyProperties.BLOCK_MODE_GCM
                )
                .setEncryptionPaddings(
                    KeyProperties.ENCRYPTION_PADDING_NONE
                )
                .setKeySize(256)
                .build()
        )

        return keyGenerator.generateKey()
    }
}

    )

    /**
     * Encrypts the random BackupKey using the password-derived KEK.
     *
     * The KEK itself is never stored.
     */
    fun wrapBackupKey(
        backupKey: ByteArray,
        kek: ByteArray
    ): WrappedBackupKey {

        require(backupKey.size == KEY_SIZE) {
            "Backup key must be 32 bytes."
        }

        require(kek.size == KEY_SIZE) {
            "KEK must be 32 bytes."
        }

        val iv = ByteArray(IV_SIZE)
        SecureRandom().nextBytes(iv)

        val cipher = Cipher.getInstance(AES_MODE)

        val secretKey =
            SecretKeySpec(kek, "AES")

        val spec =
            GCMParameterSpec(
                GCM_TAG_LENGTH,
                iv
            )

        cipher.init(
            Cipher.ENCRYPT_MODE,
            secretKey,
            spec
        )

        val encryptedKey =
            cipher.doFinal(backupKey)

        return WrappedBackupKey(
            encryptedKey =
                Base64.encodeToString(
                    encryptedKey,
                    Base64.NO_WRAP
                ),
            iv =
                Base64.encodeToString(
                    iv,
                    Base64.NO_WRAP
                )
        )
    }

    /**
     * Recovers the BackupKey using the password-derived KEK.
     *
     * An incorrect KEK/password will cause AES-GCM authentication
     * to fail rather than returning an incorrect BackupKey.
     */
    fun unwrapBackupKey(
        encryptedKey: String,
        iv: String,
        kek: ByteArray
    ): ByteArray {

        require(kek.size == KEY_SIZE) {
            "KEK must be 32 bytes."
        }

        val encryptedBytes =
            Base64.decode(
                encryptedKey,
                Base64.NO_WRAP
            )

        val ivBytes =
            Base64.decode(
                iv,
                Base64.NO_WRAP
            )

        require(ivBytes.size == IV_SIZE) {
            "Invalid IV."
        }

        val cipher =
            Cipher.getInstance(AES_MODE)

        val secretKey =
            SecretKeySpec(kek, "AES")

        val spec =
            GCMParameterSpec(
                GCM_TAG_LENGTH,
                ivBytes
            )

        cipher.init(
            Cipher.DECRYPT_MODE,
            secretKey,
            spec
        )

        val backupKey =
            cipher.doFinal(encryptedBytes)

        require(backupKey.size == KEY_SIZE) {
            "Invalid decrypted backup key."
        }

        return backupKey
    }
}

package com.example.evoke.security

import android.content.Context
import android.security.keystore.KeyGenParameterSpec
import android.security.keystore.KeyProperties
import android.util.Base64
import java.security.KeyStore
import java.security.SecureRandom
import javax.crypto.Cipher
import javax.crypto.KeyGenerator
import javax.crypto.SecretKey
import javax.crypto.spec.GCMParameterSpec

class DatabaseKeyManager(
    private val context: Context
) {

    companion object {
        private const val DATABASE_FILE_NAME = "evoke_database"
        private const val KEYSTORE_PROVIDER = "AndroidKeyStore"
        private const val MASTER_KEY_ALIAS =
            "dementia_database_master_key"

        private const val PREFS_NAME =
            "database_security"

        private const val ENCRYPTED_KEY =
            "encrypted_database_key"

        private const val IV =
            "database_key_iv"

        private const val AES_MODE =
            "AES/GCM/NoPadding"

        private const val GCM_TAG_LENGTH =
            128

        private const val DATABASE_KEY_SIZE =
            32
    }

    private val preferences =
        context.getSharedPreferences(
            PREFS_NAME,
            Context.MODE_PRIVATE
        )

    /**
     * Gets the local SQLCipher database key.
     *
     * If this is the first run, a new random 32-byte
     * database key is generated and protected using
     * Android Keystore.
     */
    fun getDatabaseKey(): ByteArray {

        val encryptedKey = preferences.getString(ENCRYPTED_KEY, null)
        val iv = preferences.getString(IV, null)

        if (encryptedKey == null && iv == null) {
            // Only a genuine first run may create a key.
            check(!context.getDatabasePath(DATABASE_FILE_NAME).exists()) {
                "A database exists but its key is missing."
            }
            val databaseKey = generateDatabaseKey()
            saveDatabaseKey(databaseKey)
            return databaseKey
        }

        check(encryptedKey != null && iv != null) {
            "The stored database key record is incomplete."
        }

        return decryptDatabaseKey(
            encryptedKey = Base64.decode(encryptedKey, Base64.DEFAULT),
            iv = Base64.decode(iv, Base64.DEFAULT)
        )
    }

    /**
     * Restores a database key obtained from a backup.
     *
     * This is used only when restoring a backup onto
     * a new device.
     *
     * The supplied database key is NOT stored directly.
     * It is encrypted using a newly generated local
     * Android Keystore master key.
     */

    fun restoreDatabaseKey(databaseKey: ByteArray) {
        require(databaseKey.size == DATABASE_KEY_SIZE) {
            "Database key must be 32 bytes."
        }
        // Keep the existing Keystore master key. saveDatabaseKey() replaces both
        // stored values in one commit(), so there is no moment with no key.
        saveDatabaseKey(databaseKey)
    }

    /**
     * Returns the current database key for the
     * optional backup route.
     *
     * This does NOT make the local key portable by itself.
     * It simply obtains the same key that SQLCipher uses.
     */
    fun getDatabaseKeyForBackup(): ByteArray {
        return getDatabaseKey()
    }


    private fun generateDatabaseKey(): ByteArray {

        val key =
            ByteArray(
                DATABASE_KEY_SIZE
            )

        SecureRandom().nextBytes(key)

        return key
    }

    private fun saveDatabaseKey(
        databaseKey: ByteArray
    ) {

        require(
            databaseKey.size == DATABASE_KEY_SIZE
        ) {
            "Database key must be 32 bytes."
        }

        val cipher =
            Cipher.getInstance(
                AES_MODE
            )

        cipher.init(
            Cipher.ENCRYPT_MODE,
            getOrCreateMasterKey()
        )

        val encrypted =
            cipher.doFinal(
                databaseKey
            )

        val iv =
            cipher.iv

        // commit() (synchronous) instead of apply(): if the process dies right
        // after a restore, the new key must already be on disk.
        val committed = preferences.edit()
            .putString(
                ENCRYPTED_KEY,
                Base64.encodeToString(
                    encrypted,
                    Base64.NO_WRAP
                )
            )
            .putString(
                IV,
                Base64.encodeToString(
                    iv,
                    Base64.NO_WRAP
                )
            )
            .commit()

        check(committed) {
            "Unable to store the database key."
        }
    }

    private fun decryptDatabaseKey(
        encryptedKey: ByteArray,
        iv: ByteArray
    ): ByteArray {

        val cipher =
            Cipher.getInstance(
                AES_MODE
            )

        val spec =
            GCMParameterSpec(
                GCM_TAG_LENGTH,
                iv
            )

        cipher.init(
            Cipher.DECRYPT_MODE,
            getOrCreateMasterKey(),
            spec
        )

        val databaseKey =
            cipher.doFinal(
                encryptedKey
            )

        require(
            databaseKey.size == DATABASE_KEY_SIZE
        ) {
            "Invalid database key."
        }

        return databaseKey
    }

    private fun getOrCreateMasterKey(): SecretKey {

        val keyStore =
            KeyStore.getInstance(
                KEYSTORE_PROVIDER
            ).apply {
                load(null)
            }

        val existingKey =
            keyStore.getKey(
                MASTER_KEY_ALIAS,
                null
            ) as? SecretKey

        if (existingKey != null) {
            return existingKey
        }

        val keyGenerator =
            KeyGenerator.getInstance(
                KeyProperties.KEY_ALGORITHM_AES,
                KEYSTORE_PROVIDER
            )

        keyGenerator.init(
            KeyGenParameterSpec.Builder(
                MASTER_KEY_ALIAS,
                KeyProperties.PURPOSE_ENCRYPT or
                        KeyProperties.PURPOSE_DECRYPT
            )
                .setBlockModes(
                    KeyProperties.BLOCK_MODE_GCM
                )
                .setEncryptionPaddings(
                    KeyProperties.ENCRYPTION_PADDING_NONE
                )
                .setKeySize(256)
                .build()
        )

        return keyGenerator.generateKey()
    }

    /**
     * Removes only the locally stored encrypted
     * database key and its Keystore master key.
     *
     * Used internally during restoration.
     */
    private fun deleteStoredDatabaseKey() {

        preferences.edit()
            .remove(ENCRYPTED_KEY)
            .remove(IV)
            .commit()

        val keyStore =
            KeyStore.getInstance(
                KEYSTORE_PROVIDER
            ).apply {
                load(null)
            }

        if (
            keyStore.containsAlias(
                MASTER_KEY_ALIAS
            )
        ) {
            keyStore.deleteEntry(
                MASTER_KEY_ALIAS
            )
        }
    }

    /**
     * Completely deletes the local database key.
     *
     * Used when clearing all local application data.
     */
    fun deleteDatabaseKey() {
        deleteStoredDatabaseKey()
    }
}

package com.example.evoke.security

import org.bouncycastle.crypto.generators.Argon2BytesGenerator
import org.bouncycastle.crypto.params.Argon2Parameters
import java.security.SecureRandom

class PasswordKeyManager {

    companion object {
        const val KDF_VERSION = 1

        private const val SALT_SIZE = 16
        private const val KEY_SIZE = 32

        // Argon2id parameters — version 1 of our backup format.
        private const val MEMORY_KB = 19 * 1024
        private const val ITERATIONS = 2
        private const val PARALLELISM = 1
    }

    data class KdfParameters(
        val version: Int,
        val salt: ByteArray,
        val memoryKb: Int,
        val iterations: Int,
        val parallelism: Int
    )

    /**
     * Used when backup is enabled for the first time.
     *
     * Generates a fresh random salt and derives a 32-byte KEK
     * from the caregiver's password.
     */
    fun createKey(
        password: CharArray
    ): Pair<ByteArray, KdfParameters> {

        require(password.isNotEmpty()) {
            "Backup password cannot be empty."
        }

        val salt =
            ByteArray(SALT_SIZE).also {
                SecureRandom().nextBytes(it)
            }

        val parameters =
            KdfParameters(
                version = KDF_VERSION,
                salt = salt,
                memoryKb = MEMORY_KB,
                iterations = ITERATIONS,
                parallelism = PARALLELISM
            )

        val kek =
            deriveKey(
                password = password,
                parameters = parameters
            )

        return kek to parameters
    }

    /**
     * Used during restore.
     *
     * The salt and KDF parameters come from backup metadata,
     * while the caregiver supplies the password.
     */
    fun deriveKey(
        password: CharArray,
        parameters: KdfParameters
    ): ByteArray {

        require(password.isNotEmpty()) {
            "Backup password cannot be empty."
        }

        require(parameters.version == KDF_VERSION) {
            "Unsupported backup KDF version: ${parameters.version}"
        }

        require(parameters.salt.isNotEmpty()) {
            "Backup salt is missing."
        }

        val argon2Parameters =
            Argon2Parameters.Builder(
                Argon2Parameters.ARGON2_id
            )
                .withVersion(
                    Argon2Parameters.ARGON2_VERSION_13
                )
                .withSalt(parameters.salt)
                .withMemoryAsKB(parameters.memoryKb)
                .withIterations(parameters.iterations)
                .withParallelism(parameters.parallelism)
                .build()

        val generator =
            Argon2BytesGenerator()

        generator.init(argon2Parameters)

        val output =
            ByteArray(KEY_SIZE)

        try {

            generator.generateBytes(
                password,
                output
            )

            return output

        } finally {

            argon2Parameters.clear()
        }
    }
}

class AppDataManager(
    private val context: Context
) {

    /**
     * Completely clears locally stored application data.
     *
     * This removes:
     * - All Room/SQLCipher database records
     * - The SQLCipher database file
     * - All locally stored profile images
     * - The encrypted database key from SharedPreferences
     * - The Android Keystore master key
     *
     * After this operation, the local database cannot be recovered.
     *
     * IMPORTANT:
     * The database must not be used again after this method
     * until the application creates a new database/key.
     */
    suspend fun clearAllLocalData() = withContext(Dispatchers.IO) {

        val applicationContext =
            context.applicationContext

        // 1. Stop access to the database.
        AppDatabase.closeDatabase()

        // 2. Delete the encrypted SQLCipher database.
        applicationContext.deleteDatabase(
            "evoke_database"
        )

        // 3. Delete all locally stored profile images.
        PersonImageStorage(applicationContext)
            .deleteAllProfileImages()

        // 3b. Delete all stored conversation audio.
        AudioStorage(applicationContext).deleteAllAudio()
        File(applicationContext.filesDir, "audio").delete()


        // 4. Delete the database key and Keystore master key.
        DatabaseKeyManager(applicationContext)
            .deleteDatabaseKey()

        // 5. Remove the now-empty people directory.
        val peopleDirectory =
            File(
                applicationContext.filesDir,
                "people"
            )

        if (peopleDirectory.exists()) {
            peopleDirectory.delete()
        }
    }
}

package com.example.evoke.storage

import android.content.Context
import java.io.File
import java.io.FileInputStream
import java.io.FileOutputStream
import java.util.UUID

/**
 * Stores filtered conversation audio in the application's private storage.
 *
 * Audio files are stored separately from the database.
 * Room stores only their relative paths in PendingConversation.audioPath.
 *
 * The original audio extension is preserved.
 * Example paths:
 * audio/8b2f...a91.m4a
 * audio/8b2f...a91.wav
 * audio/8b2f...a91.mp3
 *
 * In-progress copies are written as ".<name>.part". The leading dot means
 * they can never match AUDIO_FILE_NAME_REGEX, so they are never served,
 * never backed up and are removed by [deleteTemporaryFiles].
 */
class AudioStorage(
    context: Context
) {

    companion object {
        private const val AUDIO_DIRECTORY = "audio"
        private const val TEMPORARY_SUFFIX = ".part"

        private val AUDIO_FILE_NAME_REGEX = Regex(
            "[0-9a-fA-F]{8}-[0-9a-fA-F]{4}-[0-9a-fA-F]{4}-[0-9a-fA-F]{4}-[0-9a-fA-F]{12}(\\.[A-Za-z0-9]+)?"
        )
    }

    private val appFilesDirectory = context.filesDir

    // Creates the audio directory when it is first needed.
    private val audioDirectory: File by lazy {
        File(appFilesDirectory, AUDIO_DIRECTORY).apply {
            if (!exists()) {
                check(mkdirs() || exists()) {
                    "Unable to create audio storage directory."
                }
            }

            check(isDirectory) {
                "Audio storage path is not a directory."
            }
        }
    }

    /**
     * Copies a filtered audio file into permanent app storage.
     *
     * Call this BEFORE the Room transaction that inserts the
     * Conversation and PendingConversation records.
     *
     * Once the returned path is handed to RecognitionRepository or
     * PersonRepository, the repository owns the file: it deletes it if the
     * transaction fails or if the recording cannot be queued.
     *
     * Preserves the source file's extension.
     *
     * Returns a relative path to store in PendingConversation.audioPath.
     */
    fun saveFilteredAudio(sourceFile: File): String {

        require(sourceFile.exists() && sourceFile.isFile) {
            "Source audio file does not exist."
        }

        require(sourceFile.length() > 0) {
            "Source audio file is empty."
        }

        // Preserve the original extension, if present.
        val extension = sourceFile.extension

        require(
            extension.isEmpty() ||
                    extension.matches(Regex("[A-Za-z0-9]+"))
        ) {
            "Invalid audio file extension."
        }

        require(!extension.equals("tmp", ignoreCase = true) &&
                !extension.equals("part", ignoreCase = true)) {
            "Temporary files cannot be stored as audio."
        }

        // Generate a unique filename to avoid collisions.
        val fileName = buildString {
            append(UUID.randomUUID())

            if (extension.isNotEmpty()) {
                append(".")
                append(extension)
            }
        }

        require(AUDIO_FILE_NAME_REGEX.matches(fileName)) {
            "Invalid generated audio filename."
        }

        val destinationFile = File(audioDirectory, fileName)
        val temporaryFile = File(audioDirectory, ".$fileName$TEMPORARY_SUFFIX")

        try {

            // Copy into a temporary file first.
            FileInputStream(sourceFile).use { input ->
                FileOutputStream(temporaryFile).use { output ->
                    input.copyTo(output)
                    output.fd.sync()
                }
            }

            // Verify that the complete file was copied.
            check(temporaryFile.length() == sourceFile.length()) {
                "Audio file copy was incomplete."
            }

            // Rename the temporary file to its permanent filename.
            check(temporaryFile.renameTo(destinationFile)) {
                "Unable to finalize stored audio file."
            }

            // Return a relative path for Room.
            return "$AUDIO_DIRECTORY/$fileName"

        } catch (exception: Exception) {

            // Clean up incomplete files if copying fails.
            temporaryFile.delete()
            destinationFile.delete()

            throw exception
        }
    }

    /**
     * Resolves a stored relative path into a file inside audio/.
     *
     * Paths outside the audio directory are rejected.
     * Only UUID-based filenames with optional extensions are accepted.
     */
    fun getAudioFile(audioPath: String): File? {

        if (audioPath.isBlank()) {
            return null
        }

        // Normalize Windows-style separators.
        val normalizedPath = audioPath.replace('\\', '/')
        val expectedPrefix = "$AUDIO_DIRECTORY/"

        if (!normalizedPath.startsWith(expectedPrefix)) {
            return null
        }

        val fileName = normalizedPath.removePrefix(expectedPrefix)

        // Accept only filenames generated by this class.
        if (!isStoredAudioName(fileName)) {
            return null
        }

        val file = File(audioDirectory, fileName)

        return try {

            // Ensure the resolved file stays inside audio/.
            val directoryPath =
                audioDirectory.canonicalPath + File.separator

            val canonicalPath = file.canonicalPath

            if (canonicalPath.startsWith(directoryPath)) {
                file
            } else {
                null
            }

        } catch (exception: Exception) {
            null
        }
    }

    /**
     * Returns the stored audio file if it exists.
     */
    fun loadAudio(audioPath: String): File? {

        val file = getAudioFile(audioPath)
            ?: return null

        return if (file.exists() && file.isFile) {
            file
        } else {
            null
        }
    }

    /**
     * Deletes a stored audio file using its relative path.
     *
     * Returns true if the file was deleted or did not exist.
     */
    fun deleteAudio(audioPath: String?): Boolean {

        if (audioPath.isNullOrBlank()) {
            return true
        }

        val file = getAudioFile(audioPath)
            ?: return false

        return !file.exists() || file.delete()
    }

    /**
     * Deletes leftover in-progress copies (".<name>.part", old ".tmp").
     *
     * Call once at app start.
     */
    fun deleteTemporaryFiles() {

        val directory = File(appFilesDirectory, AUDIO_DIRECTORY)

        directory.listFiles()?.forEach { file ->
            if (
                file.isFile &&
                (file.name.startsWith(".") ||
                        file.name.endsWith(TEMPORARY_SUFFIX) ||
                        file.name.endsWith(".tmp", ignoreCase = true))
            ) {
                file.delete()
            }
        }
    }

    /**
     * Deletes audio files that no database row references.
     *
     * [referencedPaths] are the relative paths currently stored in Room
     * (PendingConversationDao.getAllAudioPaths()).
     *
     * Only call this at app start, BEFORE any new recording is saved,
     * otherwise a file that is saved but not yet committed to Room
     * could be removed.
     */
    fun deleteUnreferencedAudio(referencedPaths: Collection<String>) {

        val referencedNames = referencedPaths
            .map { it.replace('\\', '/').removePrefix("$AUDIO_DIRECTORY/") }
            .toSet()

        val directory = File(appFilesDirectory, AUDIO_DIRECTORY)

        directory.listFiles()?.forEach { file ->
            if (
                file.isFile &&
                isStoredAudioName(file.name) &&
                file.name !in referencedNames
            ) {
                file.delete()
            }
        }
    }

    /**
     * Deletes every stored audio file, including leftovers.
     *
     * Used when completely resetting local application data.
     */
    fun deleteAllAudio(): Boolean {

        val directory = File(appFilesDirectory, AUDIO_DIRECTORY)

        if (!directory.exists()) {
            return true
        }

        val files = directory.listFiles()
            ?: return false

        var successful = true

        files.filter { it.isFile }.forEach { file ->
            if (!file.delete()) {
                successful = false
            }
        }

        return successful
    }

    private fun isStoredAudioName(fileName: String): Boolean {
        return AUDIO_FILE_NAME_REGEX.matches(fileName) &&
                !fileName.endsWith(".tmp", ignoreCase = true) &&
                !fileName.endsWith(TEMPORARY_SUFFIX, ignoreCase = true)
    }
}

package com.example.evoke.storage

import android.content.Context
import android.graphics.Bitmap
import android.graphics.BitmapFactory
import android.graphics.Rect
import java.io.File
import java.io.FileOutputStream
import kotlin.math.max
import kotlin.math.min

class PersonImageStorage(
    private val context: Context
) {

    companion object {
        private const val PEOPLE_DIRECTORY = "people"

        // Profile images are only for display.
        private const val PROFILE_IMAGE_SIZE = 800

        // Good balance between quality and file size.
        private const val JPEG_QUALITY = 80

        private const val IMAGE_EXTENSION = ".jpg"
    }

    private val peopleDirectory: File by lazy {
        File(context.filesDir, PEOPLE_DIRECTORY).apply {
            if (!exists()) {
                mkdirs()
            }
        }
    }

    /**
     * Saves a profile image using the detected face bounding box.
     *
     * The original image is NOT stored.
     *
     * Processing:
     * 1. Crop around the detected face
     * 2. Add a margin around the face
     * 3. Make the crop square
     * 4. Resize to 800 x 800
     * 5. Compress as JPEG
     *
     * Returns a RELATIVE path such as:
     *
     * people/person_1.jpg
     */
    fun saveProfileImage(
        personId: Long,
        bitmap: Bitmap,
        faceBox: Rect,
        marginFraction: Float = 0.25f
    ): String {

        val croppedBitmap = cropToFace(
            bitmap = bitmap,
            faceBox = faceBox,
            marginFraction = marginFraction
        )

        val resizedBitmap = resizeBitmap(
            bitmap = croppedBitmap,
            size = PROFILE_IMAGE_SIZE
        )

        val imageFile = File(
            peopleDirectory,
            "person_$personId$IMAGE_EXTENSION"
        )

        FileOutputStream(imageFile).use { outputStream ->
            resizedBitmap.compress(
                Bitmap.CompressFormat.JPEG,
                JPEG_QUALITY,
                outputStream
            )
        }

        if (resizedBitmap !== croppedBitmap) {
            resizedBitmap.recycle()
        }

        croppedBitmap.recycle()

        // Store only the relative path in Room.
        return "$PEOPLE_DIRECTORY/${imageFile.name}"
    }

    /**
     * Loads a profile image using the relative path stored in Person.
     *
     * Example:
     *
     * people/person_1.jpg
     */
    fun loadProfileImage(
        imagePath: String
    ): Bitmap? {

        val imageFile = resolveImageFile(imagePath)
            ?: return null

        if (!imageFile.exists()) {
            return null
        }

        return BitmapFactory.decodeFile(
            imageFile.absolutePath
        )
    }

    /**
     * Deletes a profile image using the person ID.
     */
    fun deleteProfileImage(
        personId: Long
    ): Boolean {

        val imageFile = File(
            peopleDirectory,
            "person_$personId$IMAGE_EXTENSION"
        )

        return !imageFile.exists() || imageFile.delete()
    }

    /**
     * Deletes a profile image using the relative path
     * stored in Person.profileImagePath.
     */
    fun deleteProfileImage(
        imagePath: String
    ): Boolean {

        val imageFile = resolveImageFile(imagePath)
            ?: return false

        return !imageFile.exists() || imageFile.delete()
    }

    /**
     * Deletes every stored person profile image.
     *
     * Useful when clearing all local application data.
     */
    fun deleteAllProfileImages() {

        if (!peopleDirectory.exists()) {
            return
        }

        peopleDirectory.listFiles()?.forEach { file ->
            if (file.isFile) {
                file.delete()
            }
        }
    }

    /**
     * Returns the application's private people directory.
     * this function is giving JVM error uncomment at your own risk!!!
     */
//    fun getPeopleDirectory(): File {
//        return peopleDirectory
//    }

    /**
     * Converts a stored relative path into a file inside
     * the application's private people directory.
     *
     * Only paths belonging to the "people" directory are accepted.
     */
    private fun resolveImageFile(
        imagePath: String
    ): File? {

        if (imagePath.isBlank()) {
            return null
        }

        val normalizedPath =
            imagePath.replace('\\', '/')

        val expectedPrefix =
            "$PEOPLE_DIRECTORY/"

        if (!normalizedPath.startsWith(expectedPrefix)) {
            return null
        }

        val fileName =
            normalizedPath.removePrefix(expectedPrefix)

        // We only expect a filename, not another directory.
        if (
            fileName.isBlank() ||
            fileName.contains("/") ||
            fileName == "." ||
            fileName == ".."
        ) {
            return null
        }

        val imageFile =
            File(peopleDirectory, fileName)

        // Final safety check: ensure the resolved file
        // remains inside the people directory.
        val directoryPath =
            peopleDirectory.canonicalPath + File.separator

        return try {
            val canonicalPath =
                imageFile.canonicalPath

            if (canonicalPath.startsWith(directoryPath)) {
                imageFile
            } else {
                null
            }

        } catch (e: Exception) {
            null
        }
    }

    /**
     * Crops the bitmap around the detected face.
     *
     * A margin is added around the face so the result
     * is not a pixel-tight face crop.
     */
    private fun cropToFace(
        bitmap: Bitmap,
        faceBox: Rect,
        marginFraction: Float
    ): Bitmap {

        val marginX =
            (faceBox.width() * marginFraction).toInt()

        val marginY =
            (faceBox.height() * marginFraction).toInt()

        var left =
            faceBox.left - marginX

        var top =
            faceBox.top - marginY

        var right =
            faceBox.right + marginX

        var bottom =
            faceBox.bottom + marginY

        val width =
            right - left

        val height =
            bottom - top

        val side =
            max(width, height)

        val centerX =
            (left + right) / 2

        val centerY =
            (top + bottom) / 2

        left =
            centerX - side / 2

        top =
            centerY - side / 2

        right =
            left + side

        bottom =
            top + side

        // Clamp to bitmap boundaries.
        left =
            max(0, left)

        top =
            max(0, top)

        right =
            min(bitmap.width, right)

        bottom =
            min(bitmap.height, bottom)

        val cropWidth =
            right - left

        val cropHeight =
            bottom - top

        require(
            cropWidth > 0 && cropHeight > 0
        ) {
            "Invalid face crop dimensions."
        }

        return Bitmap.createBitmap(
            bitmap,
            left,
            top,
            cropWidth,
            cropHeight
        )
    }

    /**
     * Resizes the bitmap to the desired square size.
     */
    private fun resizeBitmap(
        bitmap: Bitmap,
        size: Int
    ): Bitmap {

        if (
            bitmap.width == size &&
            bitmap.height == size
        ) {
            return bitmap
        }

        return Bitmap.createScaledBitmap(
            bitmap,
            size,
            size,
            true
        )
    }

/*
* look into these before deploying :
*
* Both are genuine, but neither is serious enough to block your testing. They matter before you ship to real users.

1. Process-death recovery: genuine, narrow

I earlier described the exposed window as “a few milliseconds of renames”. That understated it. The window runs from the first rename in swapInRestoredBackup until swap.commit(), and it includes the key save and verifyRestoredDatabase, which opens an SQLCipher database and runs quick_check. That can take a second or more.

If the process dies inside that window, one of two things happens:

Killed before restoreDatabaseKey runs: the files are the restored ones, but the stored key is the old one. On the next launch getDatabase() fails to open the database, and the app is stuck.
Killed after the key is saved but before commit(): the files and key are consistent, so the app works, and the .restore_old files are just leftovers. The next restore attempt deletes them as “leftovers from an earlier interrupted restore”, which is harmless here.

The first case is the real risk. It needs Android to kill the app mid-restore, which is rare but possible if the user switches away while the restore runs.

Fix: a small journal file, written just before the first rename.

Before the swap, write restore_journal.json in filesDir. It holds the previous encrypted key strings (encrypted_database_key and database_key_iv, copied as they are) and the stage SWAPPING. They are still decryptable because the Keystore master key is now kept.
On startup, before AppDatabase.getDatabase(), check for the journal. If it exists, roll back: move the live data aside, move each .restore_old copy back, restore the old key strings, then delete the journal.
Always roll back, never try to finish a half-done restore. The backup is still on Drive, so the user can simply retry.
2. commit() ignoring failed deletes: genuine, but don’t make it throw

You’re right that the Boolean is ignored. Throwing when it’s false would be wrong, though. By the time commit() runs, the restore has already succeeded, and an exception there would reach the outer catch and show “Restore failed. Please check your backup password” for a restore that worked.

The right design, once the journal exists:

Delete the journal first. This is the real commit point. From here the restore counts as done.
Delete the .restore_old copies best-effort. Log any failure, don’t throw.
At startup, with no journal present, delete any stray *.restore_old files left behind.

Without the journal, a simpler version is to ignore the failure and rely on the next restore cleaning up leftovers, which it already does. That is acceptable but leaves old data on disk longer than it should.

Suggested order

Fix 1 is real work, around 60 lines, and it has to be tested by killing the app mid-restore (for example with adb shell am kill or a deliberate Runtime.exit between the swap and the key save). Fix 2 comes almost free with it, so do them together.

Since you’re still testing the main flow, finish those tests first. I’d add the journal before release, because the stuck-app failure is the one that can’t be recovered from the UI. Tell me when you want the code and I’ll write it against your current BackupArchiveManager and DatabaseKeyManager.
* */
